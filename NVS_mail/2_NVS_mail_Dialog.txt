Private Sub UserForm_Initialize()
    mCancelled = True
    mResult = -1
    mSel = -1
    mSelRow = -1
    mHover = -1
    mAction = "close"
    mOwner = GetForegroundWindow()      ' remembered before this window appears
End Sub

Private Sub UserForm_Activate()
    On Error Resume Next
    If Not mPlaced Then
        mPlaced = True
        Select Case mMode
            Case MODE_FOLDER
                mHwnd = NVS_PlaceForm(Me, mOwner, True)
                NVS_WheelOn Me, mHwnd
            Case MODE_LIST
                mHwnd = NVS_PlaceForm(Me, mOwner, False)
                NVS_WheelOn Me, mHwnd
            Case Else
                mHwnd = NVS_PlaceForm(Me, mOwner, False)
        End Select
    End If
    Select Case mMode
        Case MODE_FOLDER
            txtFilter.SetFocus
        Case MODE_COLOR, MODE_LIST
            txtKeys.SetFocus
        Case MODE_RENAME
            mBoxes(1).SetFocus
            mBoxes(1).SelStart = 0
            mBoxes(1).SelLength = Len(mBoxes(1).Text)
    End Select
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then
        Cancel = True
        If mMode = MODE_PROGRESS Then
            mCancelRequested = True
        Else
            mCancelled = True
            mAction = "close"
            HideNow
        End If
    End If
End Sub

Private Sub UserForm_Terminate()
    On Error Resume Next
    NVS_WheelOff Me
End Sub

Private Sub UserForm_Resize()
    If mMode = MODE_FOLDER Then LayoutFolders
End Sub

' Hides the window (which ends Me.Show) and stops sending it the mouse wheel.
Private Sub HideNow()
    On Error Resume Next
    NVS_WheelOff Me
    Me.Hide
End Sub

' The mouse wheel turned over this window (called by NVS_mail's wheel hook).
Public Sub NvsWheel(ByVal notches As Long)
    On Error Resume Next
    Select Case mMode
        Case MODE_FOLDER
            ScrollTree -3 * notches
        Case MODE_LIST
            If Not fraRows Is Nothing Then ScrollListBy -notches * mRowH * 2
    End Select
End Sub

Private Sub btnOK_Click()
    Select Case mMode
        Case MODE_FOLDER
            ChooseFolder
        Case MODE_COLOR
            If mColor >= 1 Then
                mCancelled = False
                HideNow
            Else
                Beep
            End If
        Case MODE_RENAME
            mCancelled = False
            HideNow
    End Select
End Sub

Private Sub btnCancel_Click()
    If mMode = MODE_PROGRESS Then
        mCancelRequested = True
        btnCancel.Enabled = False
        btnCancel.Caption = "Stopping..."
    Else
        mCancelled = True
        mAction = "close"
        HideNow
    End If
End Sub

Private Function ContrastText(ByVal colour As Long) As Long
    Dim r As Long, g As Long, b As Long
    r = colour And &HFF&
    g = (colour \ &H100&) And &HFF&
    b = (colour \ &H10000) And &HFF&
    If (r * 299 + g * 587 + b * 114) / 1000 > 150 Then ContrastText = RGB(0, 0, 0) Else ContrastText = RGB(255, 255, 255)
End Function

'==============================================================================
' Move Items window
'   Every folder is shown open, in the order Outlook keeps them. From the
'   Search box or the tree: Up / Down / Page Up / Page Down move, Right opens
'   a folder (again: its first subfolder), Left closes it (again: its parent),
'   Enter = OK. Type to search folders anywhere. Click an arrow to open or
'   close a folder; double-click a folder = OK. The mouse wheel scrolls. The
'   window can be resized, remembers its size and always fits the screen.
'==============================================================================

Public Function PickFolder(ByVal title As String, ByVal prompt As String, ByVal hint As String, _
                           ByVal defaultFolder As Outlook.Folder, ByVal okCaption As String) As Outlook.Folder
    Dim defID As String, i As Long, lbl As MSForms.Label, w As Long, h As Long

    mMode = MODE_FOLDER
    Me.Caption = title
    If hint = "" Then SizeForm 320, 280 Else SizeForm 320, 310

    Set lblPrompt = NewLabel(Me, 6, 6, 234, 13, prompt)
    Set lbl = NewLabel(Me, 6, 25, 38, 13, "Search:")
    lbl.Accelerator = "S"
    Set txtFilter = NewTextBox(Me, 44, 22, 196, 18)
    Set fraTree = Me.Controls.Add("Forms.Frame.1")
    With fraTree
        .Left = 6
        .Top = 44
        .Width = 234
        .Height = 230
        .Caption = ""
        .SpecialEffect = 2              ' sunken, like a list box
        .BackColor = RGB(255, 255, 255)
    End With
    Set txtKeys = NewKeyCatcher(fraTree)
    Set sbTree = fraTree.Controls.Add("Forms.ScrollBar.1")
    sbTree.Orientation = 0              ' vertical
    sbTree.TabStop = False
    sbTree.Visible = False
    Set lblHit = NewLabel(fraTree, 0, 0, 10, 10, "")

    Set btnOK = NewButton(Me, 246, 44, 68, 21, okCaption)
    btnOK.Default = True
    Set btnCancel = NewButton(Me, 246, 69, 68, 21, "Cancel")
    btnCancel.Cancel = True
    Set btnNew = NewButton(Me, 246, 94, 68, 21, "New...")
    btnNew.Accelerator = "N"
    If hint <> "" Then
        Set lblHint = NewLabel(Me, 6, 280, 308, 28, hint)
        lblHint.ForeColor = RGB(96, 96, 96)
        lblHint.Font.Size = 8
        lblHint.WordWrap = True
    End If

    ' Reopen at the size you last left it (made to fit the screen when shown).
    w = Val(GetSetting("NVS_mail", "Window", "MoveWidth", "0"))
    h = Val(GetSetting("NVS_mail", "Window", "MoveHeight", "0"))
    If w >= 240 And h >= 180 Then
        Me.Width = w
        Me.Height = h
    End If

    Set mStores = New Collection
    If Not defaultFolder Is Nothing Then
        On Error Resume Next
        mStores.Add defaultFolder.Store
        defID = defaultFolder.EntryID
        On Error GoTo 0
    End If
    NVS_BuildFolderCache mStores, False

    ' Every folder starts open, so subfolders show without opening anything.
    Set mExpanded = CreateObject("Scripting.Dictionary")
    For i = 0 To gNVS_FolderCount - 1
        If gNVS_FolderKids(i) > 0 Then mExpanded(gNVS_FolderID(i)) = True
    Next
    mSel = NVS_FolderCacheIndex(defID)
    mTreeFocus = True
    LayoutFolders
    RefreshFolders True

    Me.Show
    SaveSetting "NVS_mail", "Window", "MoveWidth", CStr(CLng(Me.Width))
    SaveSetting "NVS_mail", "Window", "MoveHeight", CStr(CLng(Me.Height))
    If Not mCancelled And mResult >= 0 Then Set PickFolder = NVS_FolderFromCache(mResult)
End Function

' Fits the controls to the window's size, and makes enough rows to fill it.
Private Sub LayoutFolders()
    Dim w As Single, h As Single, bx As Single, hintH As Single, k As Long
    If fraTree Is Nothing Or btnNew Is Nothing Then Exit Sub
    w = Me.InsideWidth
    h = Me.InsideHeight
    If w < 200 Or h < 120 Then Exit Sub
    If Not lblHint Is Nothing Then hintH = 30
    bx = w - 74
    lblPrompt.Width = bx - 12
    txtFilter.Width = bx - 6 - txtFilter.Left
    fraTree.Width = bx - 12
    fraTree.Height = h - fraTree.Top - 6 - hintH
    btnOK.Left = bx
    btnCancel.Left = bx
    btnNew.Left = bx
    If Not lblHint Is Nothing Then
        lblHint.Top = h - hintH + 2
        lblHint.Width = w - 12
    End If

    mSlots = Int((fraTree.InsideHeight - 2) / ROW_H)
    If mSlots < 1 Then mSlots = 1
    If mSlotsMade < mSlots Then
        ReDim Preserve mRowBack(0 To mSlots - 1)
        ReDim Preserve mRowArrow(0 To mSlots - 1)
        ReDim Preserve mRowName(0 To mSlots - 1)
        ReDim Preserve mRowPath(0 To mSlots - 1)
        For k = mSlotsMade To mSlots - 1
            Set mRowBack(k) = NewBlock(fraTree, 0, k * ROW_H + 1, 10, ROW_H, RGB(255, 255, 255))
            Set mRowArrow(k) = NewLabel(fraTree, 0, k * ROW_H + 3, 12, ROW_H - 3, "")
            mRowArrow(k).Font.Name = ICON_FONT
            mRowArrow(k).Font.Size = 7
            mRowArrow(k).ForeColor = RGB(96, 96, 96)
            Set mRowName(k) = NewLabel(fraTree, 0, k * ROW_H + 1, 10, ROW_H - 1, "")
            mRowName(k).WordWrap = False
            Set mRowPath(k) = NewLabel(fraTree, 0, k * ROW_H + 1, 10, ROW_H - 1, "")
            mRowPath(k).WordWrap = False
            mRowPath(k).ForeColor = RGB(120, 120, 120)
        Next
        mSlotsMade = mSlots
        lblHit.ZOrder 0                 ' the click area stays on top of the rows
    End If
    For k = mSlots To mSlotsMade - 1    ' rows that no longer fit
        mRowBack(k).Visible = False
        mRowArrow(k).Visible = False
        mRowName(k).Visible = False
        mRowPath(k).Visible = False
    Next
    sbTree.Top = 0
    sbTree.Height = fraTree.InsideHeight
    sbTree.Width = 12
    sbTree.Left = fraTree.InsideWidth - sbTree.Width
    lblHit.Top = 0
    lblHit.Height = fraTree.InsideHeight
    If mRowCount > 0 Then KeepSelVisible
    PaintTree
End Sub

Private Function Searching() As Boolean
    If Not txtFilter Is Nothing Then Searching = (Trim$(txtFilter.Text) <> "")
End Function

' Refills the rows: the folder tree, or the folders matching Search - best
' matches first (name starts with what you typed, then a word in the name,
' then anywhere in the path).
Private Sub RefreshFolders(ByVal scrollToSel As Boolean)
    Dim i As Long, n As Long, skipBelow As Long, words() As String, q As String
    Dim b As Long, cnt(0 To 4) As Long, buck() As Long, nm As String, sc As Long

    q = LCase$(Trim$(txtFilter.Text))
    ReDim mRows(0 To gNVS_FolderCount)
    If q = "" Then
        skipBelow = -1
        For i = 0 To gNVS_FolderCount - 1
            ' Folders are listed parent first, so everything deeper than a closed
            ' folder - up to the next folder at its level - is hidden.
            If skipBelow >= 0 Then
                If gNVS_FolderDepth(i) > skipBelow Then GoTo NextFolder
                skipBelow = -1
            End If
            mRows(n) = i
            n = n + 1
            If gNVS_FolderKids(i) > 0 Then
                If Not mExpanded.Exists(gNVS_FolderID(i)) Then skipBelow = gNVS_FolderDepth(i)
            End If
NextFolder:
        Next
    Else
        words = Split(q, " ")
        ReDim buck(0 To 4, 0 To gNVS_FolderCount)
        For i = 0 To gNVS_FolderCount - 1
            sc = MatchScore(i, q, words)
            If sc >= 0 Then
                buck(sc, cnt(sc)) = i
                cnt(sc) = cnt(sc) + 1
            End If
        Next
        For b = 0 To 4
            For i = 0 To cnt(b) - 1
                mRows(n) = buck(b, i)
                n = n + 1
            Next
        Next
    End If
    mRowCount = n

    mSelRow = -1
    For i = 0 To n - 1
        If mRows(i) = mSel Then
            mSelRow = i
            Exit For
        End If
    Next
    If q <> "" And n > 0 Then
        mSelRow = 0                     ' the best match
        mSel = mRows(0)
    End If
    If mTop > mRowCount - mSlots Then mTop = mRowCount - mSlots
    If mTop < 0 Or q <> "" Then mTop = 0
    If scrollToSel Then KeepSelVisible
    PaintTree
End Sub

' -1 if folder i doesn't match the search; otherwise 0 (best) to 4.
Private Function MatchScore(ByVal i As Long, ByVal q As String, words() As String) As Long
    Dim path As String, nm As String, k As Long
    path = LCase$(gNVS_FolderPath(i))
    For k = LBound(words) To UBound(words)
        If words(k) <> "" Then
            If InStr(1, path, words(k), vbBinaryCompare) = 0 Then
                MatchScore = -1
                Exit Function
            End If
        End If
    Next
    nm = LCase$(gNVS_FolderName(i))
    If Left$(nm, Len(q)) = q Then
        MatchScore = 0
    ElseIf Left$(nm, Len(words(0))) = words(0) Then
        MatchScore = 1
    ElseIf InStr(1, nm, q, vbBinaryCompare) > 0 Then
        MatchScore = 2
    ElseIf InStr(1, " " & Replace(Replace(nm, "-", " "), "_", " "), " " & words(0), vbBinaryCompare) > 0 Then
        MatchScore = 3
    Else
        MatchScore = 4
    End If
End Function

' Draws the rows on screen.
Private Sub PaintTree()
    Dim k As Long, r As Long, i As Long, x As Single, rowW As Single, srch As Boolean
    Dim maxName As Single, back As Long
    If mSlotsMade = 0 Or mPainting Then Exit Sub
    mPainting = True
    On Error Resume Next
    srch = Searching()

    sbTree.Visible = (mRowCount > mSlots)
    If sbTree.Visible Then
        sbTree.Min = 0
        sbTree.Max = mRowCount - mSlots
        sbTree.LargeChange = IIf(mSlots > 1, mSlots - 1, 1)
        sbTree.SmallChange = 1
        sbTree.Value = mTop
        rowW = sbTree.Left
    Else
        rowW = fraTree.InsideWidth
    End If
    lblHit.Width = rowW

    For k = 0 To mSlots - 1
        r = mTop + k
        If r < mRowCount Then
            i = mRows(r)
            If r = mSelRow Then
                If mTreeFocus Then back = RGB(204, 232, 255) Else back = RGB(217, 217, 217)
            ElseIf r = mHover Then
                back = RGB(229, 243, 255)
            Else
                back = RGB(255, 255, 255)
            End If
            mRowBack(k).Width = rowW
            mRowBack(k).BackColor = back
            mRowBack(k).Visible = True
            If srch Then
                mRowArrow(k).Visible = False
                x = 6
                maxName = rowW - x - 40
                mRowName(k).AutoSize = False
                mRowName(k).Left = x
                mRowName(k).Caption = gNVS_FolderName(i)
                mRowName(k).AutoSize = True
                If mRowName(k).Width > maxName Then
                    mRowName(k).AutoSize = False
                    mRowName(k).Width = maxName
                End If
                mRowName(k).Height = ROW_H - 1
                mRowPath(k).Left = x + mRowName(k).Width + 6
                mRowPath(k).Width = rowW - mRowPath(k).Left
                mRowPath(k).Caption = ParentPath(i)
                mRowPath(k).Visible = True
            Else
                x = 4 + gNVS_FolderDepth(i) * INDENT
                If gNVS_FolderKids(i) > 0 Then
                    If mExpanded.Exists(gNVS_FolderID(i)) Then
                        mRowArrow(k).Caption = ChrW$(&HE70D)     ' chevron down: open
                    Else
                        mRowArrow(k).Caption = ChrW$(&HE76C)     ' chevron right: closed
                    End If
                    mRowArrow(k).Left = x
                    mRowArrow(k).Visible = True
                Else
                    mRowArrow(k).Visible = False
                End If
                mRowName(k).AutoSize = False
                mRowName(k).Left = x + 13
                mRowName(k).Width = rowW - mRowName(k).Left
                mRowName(k).Caption = gNVS_FolderName(i)
                mRowPath(k).Visible = False
            End If
            mRowName(k).Font.Bold = (gNVS_FolderDepth(i) = 0)       ' mailbox names, as in Outlook
            mRowName(k).Visible = True
        Else
            mRowBack(k).Visible = False
            mRowArrow(k).Visible = False
            mRowName(k).Visible = False
            mRowPath(k).Visible = False
        End If
    Next
    mPainting = False
End Sub

' "Inbox > Projects" for a search result: where the folder is.
Private Function ParentPath(ByVal i As Long) As String
    Dim p As String, k As Long
    p = gNVS_FolderPath(i)
    k = InStrRev(p, "\")
    If k > 0 Then ParentPath = Replace(Left$(p, k - 1), "\", " > ")
    k = InStr(p, "   [")                                    ' another mailbox
    If k > 0 And InStr(ParentPath, "[") = 0 Then ParentPath = ParentPath & Mid$(p, k)
End Function

' Scrolls so the selected row is on screen.
Private Sub KeepSelVisible()
    If mSelRow < 0 Then Exit Sub
    If mSelRow < mTop Then mTop = mSelRow
    If mSelRow > mTop + mSlots - 1 Then mTop = mSelRow - mSlots + 1
    If mTop > mRowCount - mSlots Then mTop = mRowCount - mSlots
    If mTop < 0 Then mTop = 0
End Sub

Private Sub ScrollTree(ByVal delta As Long)
    Dim t As Long
    t = mTop + delta
    If t > mRowCount - mSlots Then t = mRowCount - mSlots
    If t < 0 Then t = 0
    If t <> mTop Then
        mTop = t
        mHover = -1
        PaintTree
    End If
End Sub

Private Sub SelectRow(ByVal r As Long)
    If mRowCount = 0 Then Exit Sub
    If r < 0 Then r = 0
    If r > mRowCount - 1 Then r = mRowCount - 1
    mSelRow = r
    mSel = mRows(r)
    KeepSelVisible
    PaintTree
End Sub

Private Sub MoveSel(ByVal delta As Long)
    If mRowCount = 0 Then Exit Sub
    If mSelRow < 0 Then
        If delta > 0 Then SelectRow mTop Else SelectRow mTop + mSlots - 1
    Else
        SelectRow mSelRow + delta
    End If
End Sub

Private Sub ToggleFolder(ByVal i As Long)
    Dim id As String
    id = gNVS_FolderID(i)
    If mExpanded.Exists(id) Then mExpanded.Remove id Else mExpanded(id) = True
    mSel = i
    RefreshFolders False
End Sub

' Right: open the folder, or if it's open already go to its first subfolder.
Private Sub OpenOrEnter()
    If mSel < 0 Or Searching() Then Exit Sub
    If gNVS_FolderKids(mSel) = 0 Then Exit Sub
    If mExpanded.Exists(gNVS_FolderID(mSel)) Then
        MoveSel 1
    Else
        ToggleFolder mSel
    End If
End Sub

' Left: close the folder, or if it's closed (or has none) go to its parent.
Private Sub CloseOrLeave()
    Dim p As Long
    If mSel < 0 Or Searching() Then Exit Sub
    If gNVS_FolderKids(mSel) > 0 And mExpanded.Exists(gNVS_FolderID(mSel)) Then
        ToggleFolder mSel
    Else
        p = gNVS_FolderParent(mSel)
        If p >= 0 Then
            mSel = p
            RefreshFolders True
        End If
    End If
End Sub

' * : opens the folder and everything inside it.
Private Sub OpenAllBelow()
    Dim i As Long, d As Long
    If mSel < 0 Or Searching() Then Exit Sub
    d = gNVS_FolderDepth(mSel)
    For i = mSel To gNVS_FolderCount - 1
        If i > mSel And gNVS_FolderDepth(i) <= d Then Exit For
        If gNVS_FolderKids(i) > 0 Then mExpanded(gNVS_FolderID(i)) = True
    Next
    RefreshFolders True
End Sub

Private Sub ChooseFolder()
    If mSel < 0 Or mSelRow < 0 Then
        Beep
        Exit Sub
    End If
    mResult = mSel
    mCancelled = False
    HideNow
End Sub

Private Sub SetTreeFocus(ByVal onIt As Boolean)
    If mTreeFocus = onIt Then Exit Sub
    mTreeFocus = onIt
    PaintTree
End Sub

Private Sub txtFilter_Change()
    Dim i As Long
    If Trim$(txtFilter.Text) = "" And mSel >= 0 Then
        ' Back to the tree: open the folders above the one picked in Search.
        i = gNVS_FolderParent(mSel)
        Do While i >= 0
            If gNVS_FolderKids(i) > 0 Then mExpanded(gNVS_FolderID(i)) = True
            i = gNVS_FolderParent(i)
        Loop
    End If
    RefreshFolders True
End Sub

    SetTreeFocus True
End Sub

Private Sub txtFilter_Exit(ByVal Cancel As MSForms.ReturnBoolean)
    SetTreeFocus False
End Sub

Private Sub txtFilter_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Select Case KeyCode
        Case vbKeyReturn
            KeyCode = 0
            ChooseFolder
        Case vbKeyDown, vbKeyUp, vbKeyPageDown, vbKeyPageUp
            TreeKey KeyCode, Shift
        Case vbKeyHome, vbKeyEnd
            If Shift = 2 Or txtFilter.Text = "" Then TreeKey KeyCode, 0      ' Ctrl+Home / Ctrl+End
        Case vbKeyRight, vbKeyLeft
            If txtFilter.Text = "" Then TreeKey KeyCode, Shift                ' nothing to move the cursor in
    End Select
End Sub

' Keys for the tree, from the tree itself or from the Search box.
Private Sub TreeKey(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Select Case KeyCode
        Case vbKeyDown: KeyCode = 0: MoveSel 1
        Case vbKeyUp: KeyCode = 0: MoveSel -1
        Case vbKeyPageDown: KeyCode = 0: MoveSel IIf(mSlots > 1, mSlots - 1, 1)
        Case vbKeyPageUp: KeyCode = 0: MoveSel -IIf(mSlots > 1, mSlots - 1, 1)
        Case vbKeyHome: KeyCode = 0: SelectRow 0
        Case vbKeyEnd: KeyCode = 0: SelectRow mRowCount - 1
        Case vbKeyRight, vbKeyAdd: KeyCode = 0: OpenOrEnter
        Case vbKeyLeft, vbKeySubtract: KeyCode = 0: CloseOrLeave
        Case vbKeyMultiply: KeyCode = 0: OpenAllBelow
        Case vbKeyBack
            KeyCode = 0
            If Searching() Then
                txtFilter.SetFocus
                txtFilter.Text = Left$(txtFilter.Text, Len(txtFilter.Text) - 1)
            Else
                CloseOrLeave
            End If
        Case vbKeyReturn: KeyCode = 0: ChooseFolder
        Case vbKeyInsert: KeyCode = 0: btnNew_Click
        Case vbKeyN
            If Shift = 2 Then                ' Ctrl+N: new folder
                KeyCode = 0
                btnNew_Click
            End If
    End Select
End Sub

Private Sub TreeMouseDown(ByVal Button As Integer, ByVal X As Single, ByVal Y As Single)
    Dim r As Long, i As Long, ax As Single
    mArrowDown = False
    r = mTop + Int(Y / ROW_H)
    If r < 0 Or r >= mRowCount Then Exit Sub
    i = mRows(r)
    If Button = 1 And Not Searching() And gNVS_FolderKids(i) > 0 Then
        ax = 4 + gNVS_FolderDepth(i) * INDENT
        If X >= ax - 4 And X <= ax + 13 Then         ' on the arrow
            mArrowDown = True
            mSel = i
            ToggleFolder i
            txtKeys.SetFocus
            Exit Sub
        End If
    End If
    mSelRow = r
    mSel = i
    txtKeys.SetFocus
    mTreeFocus = True
    PaintTree
End Sub

' New... creates a folder inside the selected one, like Outlook's Move Items.
Private Sub btnNew_Click()
    Dim parentF As Outlook.Folder, nm As String, nf As Outlook.Folder, newID As String
    If mSel < 0 Then
        Beep
        Exit Sub
    End If
    nm = Trim$(InputBox("Name of the new folder inside '" & gNVS_FolderName(mSel) & "':", "Create New Folder"))
    If nm = "" Then Exit Sub
    Set parentF = NVS_FolderFromCache(mSel)
    If parentF Is Nothing Then Exit Sub

    On Error Resume Next
    Set nf = parentF.Folders.Add(nm)
    If nf Is Nothing Then
        MsgBox "Couldn't create the folder: " & Err.Description, vbExclamation + vbMsgBoxSetForeground, "Create New Folder"
        Exit Sub
    End If
    newID = nf.EntryID
    On Error GoTo 0

    mExpanded(gNVS_FolderID(mSel)) = True
    NVS_BuildFolderCache mStores, True
    mSel = NVS_FolderCacheIndex(newID)
    txtFilter.Text = ""
    RefreshFolders True
    txtKeys.SetFocus
End Sub

'==============================================================================
' Folder Color window - pick a swatch, name the category
'   Arrow keys move between the colors, Enter = OK, Tab goes to the name.
'==============================================================================

' Returns the chosen color (1-25), or -1 if cancelled. Read CategoryName
' afterwards for the name typed in.
Public Function PickColor(ByVal title As String, ByVal folderName As String, ByVal current As Long, _
                          ByVal catName As String) As Long
    Dim i As Long, lbl As MSForms.Label

    mMode = MODE_COLOR
    Me.Caption = title
    SizeForm 244, 284

    Set lblPrompt = NewLabel(Me, 8, 6, 228, 14, "Color for '" & folderName & "' and its subfolders:")
    Set lblSel = NewBlock(Me, 0, 0, SWATCH + 6, SWATCH + 6, RGB(40, 40, 40))
    lblSel.Visible = False
    For i = 1 To 25
        Set mSw(i) = NewBlock(Me, GRID_X + ((i - 1) Mod 5) * CELL, GRID_Y + ((i - 1) \ 5) * CELL, _
                              SWATCH, SWATCH, NVS_SwatchColor(i))
        mSw(i).BorderStyle = 1
        mSw(i).BorderColor = RGB(255, 255, 255)
    Next
    Set lblHit = NewLabel(Me, GRID_X, GRID_Y, 5 * CELL, 5 * CELL, "")
    Set txtKeys = NewKeyCatcher(Me)         ' the arrow keys pick a swatch

    Set lblHint = NewLabel(Me, 8, 178, 228, 13, "")
    lblHint.TextAlign = 2               ' centered
    Set lbl = NewLabel(Me, 8, 196, 228, 13, "Category name (shown on the emails):")
    Set txtCat = NewTextBox(Me, 8, 210, 228, 18)
    txtCat.Text = catName
    Set lblPreview = NewBlock(Me, 8, 232, 228, 18, RGB(240, 240, 240))
    Set btnOK = NewButton(Me, 92, 256, 70, 22, "OK")
    btnOK.Default = True
    Set btnCancel = NewButton(Me, 166, 256, 70, 22, "Cancel")
    btnCancel.Cancel = True

    If current >= 1 And current <= 25 Then SelectSwatch current Else SelectSwatch 0

    Me.Show
    If mCancelled Then PickColor = -1 Else PickColor = mColor
End Function

Public Property Get CategoryName() As String
    If Not txtCat Is Nothing Then CategoryName = Trim$(txtCat.Text)
End Property

Private Sub SelectSwatch(ByVal i As Long)
    mColor = i
    If i >= 1 And i <= 25 Then
        lblSel.Left = mSw(i).Left - 3
        lblSel.Top = mSw(i).Top - 3
        lblSel.Visible = True
        lblHint.Caption = NVS_ColorName(i)
        lblPreview.BackColor = NVS_SwatchColor(i)
        lblPreview.ForeColor = ContrastText(NVS_SwatchColor(i))
    Else
        lblSel.Visible = False
        lblHint.Caption = "Click a color, or use the arrow keys"
        lblPreview.BackColor = RGB(240, 240, 240)
        lblPreview.ForeColor = RGB(96, 96, 96)
    End If
    UpdatePreview
End Sub

Private Sub UpdatePreview()
    If Not lblPreview Is Nothing Then lblPreview.Caption = "  " & txtCat.Text
End Sub

Private Sub txtCat_Change()
    UpdatePreview
End Sub

Private Function SwatchAt(ByVal X As Single, ByVal Y As Single) As Long
    Dim c As Long, r As Long
    If X < 0 Or Y < 0 Then Exit Function
    c = Int(X / CELL)
    r = Int(Y / CELL)
    If c > 4 Or r > 4 Then Exit Function
    SwatchAt = r * 5 + c + 1
End Function

' Arrow keys move around the 5 x 5 colors; Enter = OK.
Private Sub ColorKey(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Dim i As Long, r As Long, c As Long
    i = mColor
    If i < 1 Then i = 1
    r = (i - 1) \ 5
    c = (i - 1) Mod 5
    Select Case KeyCode
        Case vbKeyLeft: If c > 0 Then c = c - 1
        Case vbKeyRight: If c < 4 Then c = c + 1
        Case vbKeyUp: If r > 0 Then r = r - 1
        Case vbKeyDown: If r < 4 Then r = r + 1
        Case vbKeyHome: r = 0: c = 0
        Case vbKeyEnd: r = 4: c = 4
        Case vbKeyReturn
            KeyCode = 0
            btnOK_Click
            Exit Sub
        Case Else
            Exit Sub
    End Select
    KeyCode = 0
    SelectSwatch r * 5 + c + 1
End Sub

Private Sub lblHit_MouseUp(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    Dim i As Long, r As Long
    Select Case mMode
        Case MODE_COLOR
            i = SwatchAt(X, Y)
            If i > 0 Then SelectSwatch i
            txtKeys.SetFocus
        Case MODE_LIST
            r = Int(Y / mRowH) + 1
            If r >= 1 And r <= mListN Then SelectListRow r
            txtKeys.SetFocus
    End Select
End Sub

'==============================================================================
' List window (Folder Colors, Save Locations): rows with Add / Change / Remove
'   Up / Down pick a row, Enter = Change, Delete = Remove, Insert = Add.
'==============================================================================

' Returns "add", "change", "remove" or "close". SelectedRow says which row.
Public Function ShowList(ByVal title As String, ByVal prompt As String, ByVal n As Long, _
                         t1() As String, t2() As String, sw() As Long, _
                         ByVal showSwatches As Boolean, ByVal emptyText As String) As String
    Dim i As Long, y As Single, x As Single, lbl As MSForms.Label

    mMode = MODE_LIST
    mAction = "close"
    mListN = n
    mRowH = 32
    Me.Caption = title
    SizeForm 420, 292

    Set lblPrompt = NewLabel(Me, 8, 6, 404, 26, prompt)
    lblPrompt.WordWrap = True
    Set fraRows = Me.Controls.Add("Forms.Frame.1")
    With fraRows
        .Left = 8
        .Top = 34
        .Width = 320
        .Height = 250
        .Caption = ""
        .SpecialEffect = 2              ' sunken, like a list box
        .BackColor = RGB(255, 255, 255)
    End With
    Set txtKeys = NewKeyCatcher(fraRows)    ' Up / Down / Enter / Delete / Insert

    Set lblSel = NewBlock(fraRows, 0, 0, 316, mRowH, RGB(204, 232, 255))
    lblSel.Visible = False
    For i = 1 To n
        y = (i - 1) * mRowH
        x = 6
        If showSwatches Then
            Set lbl = NewBlock(fraRows, 6, y + 9, 14, 14, sw(i))
            lbl.BorderStyle = 1
            lbl.BorderColor = RGB(160, 160, 160)
            x = 28
        End If
        Set lbl = NewLabel(fraRows, x, y + 3, 296 - x, 13, t1(i))
        Set lbl = NewLabel(fraRows, x, y + 16, 296 - x, 13, t2(i))
        lbl.ForeColor = RGB(96, 96, 96)
    Next
    If n = 0 Then
        Set lbl = NewLabel(fraRows, 8, 8, 290, 40, emptyText)
        lbl.WordWrap = True
        lbl.ForeColor = RGB(96, 96, 96)
    End If
    If n * mRowH > fraRows.InsideHeight Then
        fraRows.ScrollBars = 2          ' vertical
        fraRows.ScrollHeight = n * mRowH + 2
    End If
    Set lblHit = NewLabel(fraRows, 0, 0, 316, IIf(n > 0, n * mRowH, 1), "")

    Set btnAdd = NewButton(Me, 336, 34, 76, 22, "Add...")
    btnAdd.Accelerator = "A"
    Set btnChange = NewButton(Me, 336, 60, 76, 22, "Change...")
    btnChange.Accelerator = "C"
    Set btnRemove = NewButton(Me, 336, 86, 76, 22, "Remove")
    btnRemove.Accelerator = "R"
    Set lblHint = NewLabel(Me, 336, 118, 78, 120, "Up / Down: pick" & vbCrLf & "Enter: Change" & vbCrLf & _
                           "Delete: Remove" & vbCrLf & "Insert: Add" & vbCrLf & "Esc: Close")
    lblHint.WordWrap = True
    lblHint.Font.Size = 8
    lblHint.ForeColor = RGB(120, 120, 120)
    Set btnCancel = NewButton(Me, 336, 262, 76, 22, "Close")
    btnCancel.Cancel = True
    If n > 0 Then SelectListRow 1 Else UpdateListButtons

    Me.Show
    ShowList = mAction
End Function

Public Property Get SelectedRow() As Long
    SelectedRow = mListSel
End Property

Private Sub SelectListRow(ByVal r As Long)
    Dim rowTop As Single
    If r < 1 Or r > mListN Then Exit Sub
    mListSel = r
    lblSel.Top = (r - 1) * mRowH
    lblSel.Visible = True
    ' Scroll so the row is on screen.
    If fraRows.ScrollBars = 2 Then
        rowTop = (r - 1) * mRowH
        If rowTop < fraRows.ScrollTop Then fraRows.ScrollTop = rowTop
        If rowTop + mRowH > fraRows.ScrollTop + fraRows.InsideHeight Then
            fraRows.ScrollTop = rowTop + mRowH - fraRows.InsideHeight
        End If
    End If
    UpdateListButtons
End Sub

Private Sub ScrollListBy(ByVal delta As Single)
    Dim t As Single
    If fraRows.ScrollBars <> 2 Then Exit Sub
    t = fraRows.ScrollTop + delta
    If t > fraRows.ScrollHeight - fraRows.InsideHeight Then t = fraRows.ScrollHeight - fraRows.InsideHeight
    If t < 0 Then t = 0
    fraRows.ScrollTop = t
End Sub

Private Sub ListKey(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Dim page As Long
    page = Int(fraRows.InsideHeight / mRowH) - 1
    If page < 1 Then page = 1
    Select Case KeyCode
        Case vbKeyDown: SelectListRow mListSel + 1
        Case vbKeyUp: SelectListRow mListSel - 1
        Case vbKeyPageDown: SelectListRow IIf(mListSel + page > mListN, mListN, mListSel + page)
        Case vbKeyPageUp: SelectListRow IIf(mListSel - page < 1, 1, mListSel - page)
        Case vbKeyHome: SelectListRow 1
        Case vbKeyEnd: SelectListRow mListN
        Case vbKeyReturn: btnChange_Click
        Case vbKeyDelete: btnRemove_Click
        Case vbKeyInsert: btnAdd_Click
        Case Else: Exit Sub
    End Select
    KeyCode = 0
End Sub

Private Sub UpdateListButtons()
    btnChange.Enabled = (mListSel >= 1)
    btnRemove.Enabled = (mListSel >= 1)
End Sub

Private Sub btnAdd_Click()
    mAction = "add"
    HideNow
End Sub

Private Sub btnChange_Click()
    If mListSel < 1 Then Exit Sub
    mAction = "change"
    HideNow
End Sub

Private Sub btnRemove_Click()
    If mListSel < 1 Then Exit Sub
    mAction = "remove"
    HideNow
End Sub

'==============================================================================
' Rename Attachments window - one box per attachment
'==============================================================================

' names() holds the names without extension; edited in place. Returns False
' if cancelled.
Public Function RenameList(ByVal n As Long, names() As String, exts() As String) As Boolean
    Dim i As Long, y As Single, x0 As Single, y0 As Single, listH As Single, rowsH As Single
    Dim host As Object, lbl As MSForms.Label

    mMode = MODE_RENAME
    Me.Caption = "Rename Attachments"
    rowsH = n * 24
    listH = rowsH
    If listH > 216 Then listH = 216
    SizeForm 380, 34 + listH + 38

    If n = 1 Then
        Set lblPrompt = NewLabel(Me, 8, 8, 364, 24, "New name for the attachment (the extension stays the same):")
    Else
        Set lblPrompt = NewLabel(Me, 8, 8, 364, 24, "Change any of the names (the extensions stay the same). Tab moves to the next one:")
    End If
    lblPrompt.WordWrap = True

    If rowsH > listH Then
        Set fraRows = Me.Controls.Add("Forms.Frame.1")
        With fraRows
            .Left = 8
            .Top = 34
            .Width = 364
            .Height = listH
            .Caption = ""
            .SpecialEffect = 0
            .ScrollBars = 2             ' vertical
            .ScrollHeight = rowsH + 4
        End With
        Set host = fraRows
        x0 = 0
        y0 = 2
    Else
        Set host = Me
        x0 = 8
        y0 = 34
    End If

    ReDim mBoxes(1 To n)
    For i = 1 To n
        y = y0 + (i - 1) * 24
        Set mBoxes(i) = NewTextBox(host, x0, y, 288, 19)
        mBoxes(i).Text = names(i)
        Set lbl = NewLabel(host, x0 + 292, y + 3, 60, 14, exts(i))
        lbl.ForeColor = RGB(96, 96, 96)
    Next

    Set btnOK = NewButton(Me, 222, 34 + listH + 10, 72, 22, "Rename")
    btnOK.Default = True
    Set btnCancel = NewButton(Me, 300, 34 + listH + 10, 72, 22, "Cancel")
    btnCancel.Cancel = True

    Me.Show
    If mCancelled Then Exit Function
    For i = 1 To n
        names(i) = mBoxes(i).Text
    Next
    RenameList = True
End Function

' Cheap to call often: repaints (and lets Outlook breathe) a few times a second.
Public Sub ProgressUpdate(ByVal done As Long, ByVal total As Long, ByVal recolored As Long)
    Dim t As Single
    t = Timer
    If t >= mLastPaint And t - mLastPaint < 0.15 Then Exit Sub
    mLastPaint = t
    If total > 0 Then
        If done >= total Then lblBar.Width = mBarW Else lblBar.Width = 1 + (mBarW - 1) * done / total
    End If
    lblHint.Caption = done & " of " & total & " emails checked - " & recolored & " recolored"
    Me.Repaint
    DoEvents
End Sub
