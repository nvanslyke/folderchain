' in the code window (such as "Option Explicit"), then paste this whole file.
'
' Watches one folder (Sent Items or a colored folder) and colors the chain
' of any email that lands in it.
'==============================================================================
Option Explicit

Private WithEvents mItems As Outlook.Items

Public Sub Attach(ByVal f As Outlook.Folder)
    Set mItems = f.Items
End Sub

Private Sub mItems_ItemAdd(ByVal Item As Object)
    On Error Resume Next
    If gNVS_Busy Then Exit Sub
    If TypeOf Item Is Outlook.MailItem Then NVS_RecolorConversation Item
End Sub
