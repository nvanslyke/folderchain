' NVS_mail - optional file 5:  NVS_mail_FindTime   (UserForm)
'
' Find a Time: when everyone on an email chain is free, from the keyboard.
' Only the people inside your firm are shown - theirs are the calendars
' Outlook can see. Pick a time and press Enter: Outlook's meeting invite
' opens with everyone on it. Nothing is ever sent: you check the invite, add
' a Teams link and send it yourself.
'
' HOW TO ADD (once): in the VBA editor (Alt+F11) choose Insert > UserForm.
' Press F4 and change (Name) to  NVS_mail_FindTime  . Right-click the form >
' View Code, delete anything already there, then paste this whole file.
' Leave the form blank - this code builds the window itself. Then add the
' macro NVS_FindTime (it's in file 1) to the ribbon or Quick Access Toolbar.
'
' Keys (also shown at the bottom of the window):
'   Up / Down          pick a time; Page Up / Page Down: the day before / after
'   Left / Right       start the meeting 15 minutes earlier / later
'   Enter              open the invite for that time
'   Ctrl+C             copy the free times, to paste into a reply
'   type a name        add someone (Enter adds them)
'   Alt+T              their times: paste the times someone sent you
'   Alt+P              people: Space = needed / optional, Delete = remove
'   Alt+L              meeting length          Alt+O  also invite people outside the firm
'   Ctrl+K             address book            F5  check the calendars again       Esc  close
'==============================================================================

' Called by NVS_FindTime (file 1). people: a Collection of Array(name,
' address, "R" or "O") from the chain. topic: the invite's subject.
Public Sub StartFor(ByVal people As Collection, ByVal topic As String, ByVal owner As LongPtr)
    Dim v As Variant, keep As Boolean
    On Error GoTo Fail
    mOwner = owner
    If Not mBuilt Then BuildUI

    ' The same chain again within half an hour: carry on where you left off.
    keep = False
    If mPN > 0 And topic = mTopic Then
        If Me.Visible Then keep = True Else keep = (mLastDay = Date And Elapsed(mLastShown) < 1800)
    End If
    mTopic = topic
    If Not keep Then
        mGen = mGen + 1                     ' a calendar being read for the last chain is dropped
        mPN = 0
        ReDim mPName(1 To 8)
        ReDim mPAddr(1 To 8)
        ReDim mPRole(1 To 8)
        ReDim mPFB(1 To 8)
        ReDim mPFBDay(1 To 8)
        ReDim mPState(1 To 8)
        mON = 0
        ReDim mOName(1 To 8)
        ReDim mOAddr(1 To 8)
        ReDim mORole(1 To 8)
        AddMe
        For Each v In people                ' needed people first, then optional
            If CStr(v(2)) = "R" Then AddChainPerson CStr(v(0)), CStr(v(1)), "R"
        Next
        For Each v In people
            If CStr(v(2)) <> "R" Then AddChainPerson CStr(v(0)), CStr(v(1)), "O"
        Next
        mIgnore = True
        txtTheir.Text = ""
        txtAdd.Text = ""
        chkOut.Value = FT_INVITE_OUTSIDE
        mIgnore = False
        mWinN = 0
        mPSel = 1
        mPTop = 1
        mRSel = 0
        mRTop = 1
    End If
    HideSug
    ShowOutside
    MakeDays
    ApplyTheir
    If topic = "" Then Me.Caption = "Find a Time" Else Me.Caption = "Find a Time - " & Left$(topic, 80)

    If Not Me.Visible Then
        Me.Show vbModeless
        mHwnd = NVS_PlaceForm(Me, owner, True, True)     ' fits the screen, comes to the front
        NVS_WheelOn Me, mHwnd
    Else
        Me.Show vbModeless
    End If
    Arrange                                 ' again once placed: that screen may be scaled differently
    BuildRows
    If mPN > 1 Or mWinN > 0 Then FocusList 2 Else txtAdd.SetFocus
    LoadCalendars
    Exit Sub
Fail:
    MsgBox "Find a Time couldn't open: " & Err.Description, vbExclamation + vbMsgBoxSetForeground, "NVS_mail"
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then
        Cancel = True                       ' kept for next time, calendars and all
        HideWindow
    End If
End Sub

Private Sub UserForm_Terminate()
    On Error Resume Next
    NVS_WheelOff Me
End Sub

Private Sub UserForm_Resize()
    If mBuilt Then Arrange
End Sub

Private Sub HideWindow()
    On Error Resume Next
    SaveSetting "NVS_mail", "Window", "FindTime2Width", CStr(CLng(Me.Width))
    SaveSetting "NVS_mail", "Window", "FindTime2Height", CStr(CLng(Me.Height))
    HideSug
    mLastShown = Timer
    mLastDay = Date
    NVS_WheelOff Me
    Me.Hide
End Sub

' The mouse wheel turned over this window (called by NVS_mail's wheel hook):
' it scrolls the list under the pointer.
Public Sub NvsWheel(ByVal notches As Long)
    On Error Resume Next
    If mOver = 1 Then ScrollPeople -3 * notches Else ScrollTimes -3 * notches
End Sub

' Fits everything to the window's size: the people on the left, the times
' on the right, and as many rows in each as there's room for.
Private Sub Arrange()
    Dim w As Single, h As Single, lw As Single, rx As Single, rw As Single, bottom As Single
    Dim outH As Single, listTop As Single

    If Not mBuilt Then Exit Sub
    On Error Resume Next
    w = Me.InsideWidth
    h = Me.InsideHeight
    If w < 520 Then w = 520
    If h < 300 Then h = 300
    If w >= 640 Then lw = 220 Else lw = 190
    rx = 8 + lw + 12
    rw = w - rx - 8
    bottom = h - 54                         ' top of the bottom area
    listTop = 84

    ' Left
    lblAdd.Move 8, 6, lw, 12
    txtAdd.Move 8, 19, lw, 18
    lblAddHint.Move 8, 40, lw, 12
    lblPeople.Move 8, 68, lw, 12
    If mON > 0 Then outH = 44
    fraPeople.Move 8, listTop, lw, bottom - listTop - 4 - outH
    chkOut.Move 8, bottom - outH + 2, lw, 16
    lblOutNames.Move 22, bottom - outH + 18, lw - 14, 22
    chkOut.Visible = (mON > 0)
    lblOutNames.Visible = (mON > 0)

    ' Right
    lblTheir.Move rx, 6, rw, 12
    txtTheir.Move rx, 19, rw, 30
    lblTheirInfo.Move rx, 50, rw, 11
    lblTimes.Move rx, 68
    cboLen.Move rx + rw - 66, 64, 66, 18
    lblLen.Move rx + rw - 66 - 40, 67, 36, 12
    fraTimes.Move rx, listTop, rw, bottom - listTop - 4

    ' Bottom
    btnInvite.Move w - 8 - 84, bottom + 2, 84, 22
    btnCopy.Move w - 8 - 84 - 6 - 84, bottom + 2, 84, 22
    lblStatus.Move 8, bottom + 7, w - 16 - 180, 13
    lblKeys.Move 8, bottom + 29, w - 16, 24

    MakePeopleRows
    MakeTimeRows
    If fraSug.Visible Then ShowSugRows
    KeepPersonVisible
    KeepRowVisible
    PaintPeople
    PaintTimes
End Sub

' Rows for the people list, as many as fit (made once, kept after).
Private Sub MakePeopleRows()
    Dim k As Long
    mPSlots = Int((fraPeople.InsideHeight - 2) / ROW_H)
    If mPSlots < 1 Then mPSlots = 1
    If mPSlotsMade < mPSlots Then
        ReDim Preserve mPRowBack(0 To mPSlots - 1)
        ReDim Preserve mPRowName(0 To mPSlots - 1)
        ReDim Preserve mPRowTag(0 To mPSlots - 1)
        ReDim Preserve mPRowStat(0 To mPSlots - 1)
        For k = mPSlotsMade To mPSlots - 1
            Set mPRowBack(k) = NewLabel(fraPeople, 0, k * ROW_H + 1, 10, ROW_H, "")
            mPRowBack(k).BackStyle = 1
            Set mPRowName(k) = NewLabel(fraPeople, 4, k * ROW_H + 2, 10, ROW_H - 2, "")
            mPRowName(k).WordWrap = False
            Set mPRowTag(k) = NewLabel(fraPeople, 0, k * ROW_H + 3, 40, ROW_H - 3, "optional")
            mPRowTag(k).WordWrap = False
            mPRowTag(k).Font.Size = 8
            mPRowTag(k).ForeColor = C_GRAY
            Set mPRowStat(k) = NewLabel(fraPeople, 0, k * ROW_H + 2, 54, ROW_H - 2, "")
            mPRowStat(k).WordWrap = False
        Next
        mPSlotsMade = mPSlots
        lblPeopleHit.ZOrder 0               ' the click area stays on top of the rows
    End If
    For k = mPSlots To mPSlotsMade - 1      ' rows that no longer fit
        mPRowBack(k).Visible = False
        mPRowName(k).Visible = False
        mPRowTag(k).Visible = False
        mPRowStat(k).Visible = False
    Next
    sbPeople.Move fraPeople.InsideWidth - 12, 0, 12, fraPeople.InsideHeight
    lblPeopleHit.Move 0, 0, fraPeople.InsideWidth, fraPeople.InsideHeight
End Sub

' You: always person 1, always looked at.
Private Sub AddMe()
    Dim addr As String
    addr = NVS_MyAddress()
    If addr = "" Then addr = MyName()
    mPN = 1
    mPName(1) = "You"
    mPAddr(1) = addr
    mPRole(1) = "R"
    mPFB(1) = ""
    mPState(1) = 0
    UseCachedCalendar 1
End Sub

' Someone from the chain: shown if they're inside the firm, otherwise kept
' for "Also invite".
Private Sub AddChainPerson(ByVal nm As String, ByVal addr As String, ByVal role As String)
    Dim i As Long
    addr = Trim$(addr)
    If addr = "" Then Exit Sub
    If NVS_IsMe(addr) Then Exit Sub
    If IsInside(addr) Then
        AddPersonRow nm, addr, role, True
        Exit Sub
    End If
    For i = 1 To mON
        If StrComp(mOAddr(i), addr, vbTextCompare) = 0 Then
            If role = "R" Then mORole(i) = "R"
            Exit Sub
        End If
    Next
    mON = mON + 1
    If mON > UBound(mOName) Then
        ReDim Preserve mOName(1 To 2 * mON)
        ReDim Preserve mOAddr(1 To 2 * mON)
        ReDim Preserve mORole(1 To 2 * mON)
    End If
    If Trim$(nm) = "" Then nm = addr
    mOName(mON) = Trim$(nm)
    mOAddr(mON) = addr
    mORole(mON) = role
End Sub

' Adds someone to the people list (or, if they're on it already, makes them
' needed if asked). Returns their row, or 0 if they weren't added.
Private Function AddPersonRow(ByVal nm As String, ByVal addr As String, ByVal role As String, _
                              ByVal inside As Boolean) As Long
    Dim i As Long
    addr = Trim$(addr)
    nm = Trim$(nm)
    If addr = "" Then Exit Function
    If Len(nm) > 2 And Left$(nm, 1) = "'" And Right$(nm, 1) = "'" Then nm = Mid$(nm, 2, Len(nm) - 2)
    If nm = "" Then nm = addr
    i = PersonIndex(addr)
    If i > 0 Then
        If role = "R" Then mPRole(i) = "R"
        AddPersonRow = i
        Exit Function
    End If
    If NVS_IsMe(addr) Then Exit Function
    mPN = mPN + 1
    If mPN > UBound(mPName) Then
        ReDim Preserve mPName(1 To 2 * mPN)
        ReDim Preserve mPAddr(1 To 2 * mPN)
        ReDim Preserve mPRole(1 To 2 * mPN)
        ReDim Preserve mPFB(1 To 2 * mPN)
        ReDim Preserve mPFBDay(1 To 2 * mPN)
        ReDim Preserve mPState(1 To 2 * mPN)
    End If
    mPName(mPN) = nm
    mPAddr(mPN) = addr
    mPRole(mPN) = role
    mPFB(mPN) = ""
    If inside Then
        mPState(mPN) = 0
        UseCachedCalendar mPN
    Else
        mPState(mPN) = 3                    ' outside the firm: invited, calendar not looked for
    End If
    AddPersonRow = mPN
End Function

' A calendar read in the last ten minutes is used again.
Private Sub UseCachedCalendar(ByVal i As Long)
    Dim k As String, v As Variant
    k = LCase$(mPAddr(i))
    If Not mFBCache.Exists(k) Then Exit Sub
    v = mFBCache(k)
    If Elapsed(CSng(v(2))) < 600 And CDbl(v(3)) = CDbl(Date) Then
        mPFB(i) = v(0)
        mPState(i) = v(1)
        mPFBDay(i) = CDate(v(3))
    End If
End Sub

Private Function PersonIndex(ByVal addr As String) As Long
    Dim i As Long
    For i = 1 To mPN
        If StrComp(mPAddr(i), addr, vbTextCompare) = 0 Then
            PersonIndex = i
            Exit Function
        End If
    Next
End Function

Private Sub RemovePerson(ByVal i As Long)
    Dim j As Long
    If i < 1 Or i > mPN Then Exit Sub
    If i = 1 Then
        Beep                                ' you're always in it
        SetStatus "You're always in it - Space makes you optional."
        Exit Sub
    End If
    For j = i To mPN - 1
        mPName(j) = mPName(j + 1)
        mPAddr(j) = mPAddr(j + 1)
        mPRole(j) = mPRole(j + 1)
        mPFB(j) = mPFB(j + 1)
        mPFBDay(j) = mPFBDay(j + 1)
        mPState(j) = mPState(j + 1)
    Next
    mPN = mPN - 1
    If mPSel > mPN Then mPSel = mPN
    KeepPersonVisible
    BuildRows
End Sub

' Space on a person: needed <-> optional. Optional people are invited, but
' the times don't wait for them.
Private Sub ToggleRole(ByVal i As Long)
    If i < 1 Or i > mPN Then Exit Sub
    If mPRole(i) = "O" Then mPRole(i) = "R" Else mPRole(i) = "O"
    BuildRows
End Sub

' True if this address is inside your firm: your domain (or one in
' FT_INSIDE_DOMAINS), or a mailbox in your firm's address book.
Private Function IsInside(ByVal addr As String) As Boolean
    Dim k As String, dom As String, r As Outlook.Recipient
    k = LCase$(Trim$(addr))
    If mInside.Exists(k) Then
        IsInside = mInside(k)
        Exit Function
    End If
    dom = DomainOf(k)
    If dom = "" Then
        IsInside = True                     ' an address book name, not an email address
    ElseIf InsideDomain(dom) Then
        IsInside = True
    Else
        On Error Resume Next
        Set r = Application.Session.CreateRecipient(addr)
        r.Resolve
        If r.Resolved Then IsInside = (r.AddressEntry.AddressEntryUserType = olExchangeUserAddressEntry)
        On Error GoTo 0
    End If
    mInside(k) = IsInside
End Function

Private Function DomainOf(ByVal addr As String) As String
    Dim p As Long
    p = InStrRev(addr, "@")
    If p > 0 Then DomainOf = LCase$(Trim$(Mid$(addr, p + 1)))
End Function

' Your domain or one of FT_INSIDE_DOMAINS, or a part of one ("ny.bank.com"
' and "bank.com" go together).
Private Function InsideDomain(ByVal dom As String) As Boolean
    Dim d As Variant, mine As String
    mine = DomainOf(NVS_MyAddress())
    If mine <> "" Then
        If SameFirm(dom, mine) Then
            InsideDomain = True
            Exit Function
        End If
    End If
    For Each d In Split(Replace(Replace(FT_INSIDE_DOMAINS, ";", ","), " ", ""), ",")
        If CStr(d) <> "" Then
            If SameFirm(dom, LCase$(Replace(CStr(d), "@", ""))) Then
                InsideDomain = True
                Exit Function
            End If
        End If
    Next
End Function

Private Function SameFirm(ByVal a As String, ByVal b As String) As Boolean
    If a = "" Or b = "" Then Exit Function
    SameFirm = (a = b) Or (Right$(a, Len(b) + 1) = "." & b) Or (Right$(b, Len(a) + 1) = "." & a)
End Function

' Enter in Add someone: the highlighted suggestion, or whoever the typed text
' finds in the address book. role "R" (needed) or "O" (optional).
Private Sub AddFromBox(ByVal role As String)

    txt = Trim$(txtAdd.Text)
    If fraSug.Visible And mSugSel >= 1 And mSugSel <= mSugN Then

        If addr = "" Then addr = nm
    ElseIf txt = "" Then
        FocusList 2
        Exit Sub

    End If

    On Error Resume Next
    Set r = Application.Session.CreateRecipient(addr)
    r.Resolve

        If r.Resolved Then
            AddResolved r, role
            Exit Sub
        End If
    End If
    If InStr(txt, "@") > 0 And InStr(txt, " ") = 0 Then
        AddTyped txt, txt, role             ' someone not in any address book

'------------------------------------------------------------------------------

' People from your recent mail whose name has words starting with what you
' typed, the ones you email most first.
Private Sub Suggest()
    Dim q As String, words() As String, k As Variant, v As Variant, n As Long, i As Long, j As Long
    Dim nm() As String, ad() As String, ct() As Long, best As Long

    q = LCase$(Trim$(txtAdd.Text))
    If Len(q) < 2 Then
        HideSug
        Exit Sub
    End If
    If mNames Is Nothing Then LoadNames
    words = Split(q, " ")
    ReDim nm(1 To 400)
    ReDim ad(1 To 400)
    ReDim ct(1 To 400)
    For Each k In mNames.Keys
        v = mNames(k)
        If NameMatches(CStr(k), CStr(v(1)), q, words) And Not AlreadyIn(CStr(v(0)), CStr(v(1))) Then
            n = n + 1
            nm(n) = v(0)
            ad(n) = v(1)
            ct(n) = v(2)
            If n = 400 Then Exit For
        End If
    Next
    If n = 0 Then
        HideSug
        Exit Sub
    End If
    mSugN = n
    If mSugN > NAME_ROWS Then mSugN = NAME_ROWS
    For i = 1 To mSugN                      ' the most emailed first
        best = 0
        For j = 1 To n
            If ct(j) >= 0 Then
                If best = 0 Then
                    best = j
                ElseIf ct(j) > ct(best) Then
                    best = j
                End If
            End If
        Next
        mSugName(i) = nm(best)
        mSugAddr(i) = ad(best)
        ct(best) = -1
    Next
    mSugSel = 1
    ShowSugRows
End Sub

            mSRowName(i).Caption = mSugName(i)

            mSRowBack(i).Visible = True
            mSRowName(i).Visible = True

End Sub

Private Function AlreadyIn(ByVal nm As String, ByVal addr As String) As Boolean
    Dim i As Long
    For i = 2 To mPN
        If StrComp(mPName(i), nm, vbTextCompare) = 0 Then AlreadyIn = True
        If addr <> "" Then
            If StrComp(mPAddr(i), addr, vbTextCompare) = 0 Then AlreadyIn = True
        End If
        If AlreadyIn Then Exit Function
    Next
End Function

' Every word typed starts a word of the name ("jo sm" finds "Smith, John"),
' or the address contains what was typed.
Private Function NameMatches(ByVal key As String, ByVal addr As String, ByVal q As String, words() As String) As Boolean
    Dim padded As String, i As Long
    If InStr(q, "@") > 0 Then
        NameMatches = (InStr(1, addr, q, vbTextCompare) > 0)
        Exit Function
    End If
    padded = " " & Replace(Replace(Replace(Replace(key, ",", " "), ".", " "), "(", " "), "-", " ")
    For i = 0 To UBound(words)
        If words(i) <> "" Then
            If InStr(padded, " " & words(i)) = 0 Then Exit Function
        End If
    Next
    NameMatches = True
End Function

' Names (and, for senders, addresses) from the last year of Sent Items and
' the last half year of the Inbox, counted, so people you email most come
' first. Read once per Outlook session.
Private Sub LoadNames()
    Set mNames = CreateObject("Scripting.Dictionary")
    SetStatus "Getting names from your recent mail..."
    DoEvents
    On Error Resume Next
    NamesFrom Application.Session.GetDefaultFolder(olFolderSentMail), "SentOn", 365, False
    NamesFrom Application.Session.GetDefaultFolder(olFolderInbox), "ReceivedTime", 180, True
    On Error GoTo 0
    SetStatus mNames.Count & " names from your recent mail."
End Sub

Private Sub NamesFrom(ByVal f As Outlook.Folder, ByVal dateField As String, ByVal days As Long, _
                      ByVal senders As Boolean)
    Dim tbl As Outlook.Table, crit As String, rw As Outlook.Row, n As Long, p As Variant
    On Error GoTo Done
    crit = "[" & dateField & "] >= '" & Format$(Date - days, "ddddd") & " 12:00 AM'"
    Set tbl = f.GetTable(crit)
    tbl.Sort "[" & dateField & "]", True
    tbl.Columns.RemoveAll
    If senders Then
        tbl.Columns.Add "SenderName"
        tbl.Columns.Add "SenderEmailAddress"
    Else
        tbl.Columns.Add "To"
        tbl.Columns.Add "CC"
    End If
    Do Until tbl.EndOfTable
        Set rw = tbl.GetNextRow
        If rw Is Nothing Then Exit Do
        n = n + 1
        If n > 6000 Then Exit Do
        If senders Then
            AddName CStr(rw(1) & ""), CStr(rw(2) & ""), 1
        Else
            For Each p In Split(rw(1) & "", ";")
                AddName CStr(p), "", 3
            Next
            For Each p In Split(rw(2) & "", ";")
                AddName CStr(p), "", 2
            Next
        End If
        If n Mod 500 = 0 Then DoEvents
    Loop
Done:
End Sub

Private Sub AddName(ByVal nm As String, ByVal addr As String, ByVal weight As Long)
    Dim k As String, v As Variant
    nm = Trim$(Replace(nm, """", ""))
    If Len(nm) > 2 And Left$(nm, 1) = "'" And Right$(nm, 1) = "'" Then nm = Mid$(nm, 2, Len(nm) - 2)
    If Len(nm) < 2 Then Exit Sub
    If StrComp(nm, MyName(), vbTextCompare) = 0 Then Exit Sub
    If addr <> "" Then
        If NVS_IsMe(addr) Then Exit Sub
    End If
    k = LCase$(nm)
    If mNames.Exists(k) Then
        v = mNames(k)
        If CStr(v(1)) = "" And addr <> "" Then v(1) = addr
        v(2) = v(2) + weight
        mNames(k) = v
    Else
        mNames.Add k, Array(nm, addr, weight)
    End If
End Sub

Private Function MyName() As String
    Static nm As String
    If nm = "" Then
        On Error Resume Next
        nm = Application.Session.CurrentUser.Name
    End If
    MyName = nm
End Function

' "Jane" from "Jane Smith" or "Smith, Jane" - or the whole name, if someone
' else on the list has the same first name. "you" for you.
Private Function ShortName(ByVal i As Long) As String
    Dim j As Long, fn As String
    If i = 1 Then
        ShortName = "you"
        Exit Function
    End If
    fn = FirstName(mPName(i))
    For j = 2 To mPN
        If j <> i Then
            If StrComp(FirstName(mPName(j)), fn, vbTextCompare) = 0 Then
                ShortName = mPName(i)
                Exit Function
            End If
        End If
    Next
    ShortName = fn
End Function

Private Function FirstName(ByVal nm As String) As String
    Dim k As Long
    nm = Trim$(nm)
    k = InStr(nm, ",")
    If k > 0 Then
        FirstName = Trim$(Split(Trim$(Mid$(nm, k + 1)) & " ", " ")(0))
    Else
        FirstName = Split(nm & " ", " ")(0)
    End If
    If FirstName = "" Then FirstName = nm
End Function

'==============================================================================
' Days and calendars
'==============================================================================

' The days looked at: FT_DAYS weekdays from today.
Private Sub MakeDays()
    Dim d As Date, n As Long
    ReDim mDay(1 To FT_DAYS + 1)
    d = Date
    Do While n < FT_DAYS
        If FT_WEEKENDS Or Weekday(d, vbMonday) <= 5 Then
            n = n + 1
            mDay(n) = d
        End If
        d = d + 1
    Loop
    mDayN = n
End Sub

' Times are counted in quarter hours from midnight today (mBase).
Private Function QTime(ByVal q As Long) As Date
    QTime = mBase + q / 96#
End Function

' The quarter hour of a time: roundUp = the next one, unless it's on one.
Private Function QOf(ByVal t As Date, ByVal roundUp As Boolean) As Long
    Dim x As Double
    x = (CDbl(t) - CDbl(mBase)) * 96#
    If roundUp Then QOf = -Int(-(x - 0.01)) Else QOf = Int(x + 0.01)
End Function

Private Function DayQ(ByVal d As Long) As Long
    DayQ = CLng((mDay(d) - mBase) * 96)
End Function

Private Function DayOfQ(ByVal q As Long) As Date
    DayOfQ = mBase + Int(q / 96)
End Function

' Person p in quarter hour q: 0 free, 1 tentative, 2 busy, 3 away (out of
' office), 4 working elsewhere; -1 if it isn't known.
Private Function Code(ByVal p As Long, ByVal q As Long) As Long
    Dim u As Long
    Code = -1
    If mPState(p) <> 1 Then Exit Function
    u = q - CLng((mPFBDay(p) - mBase) * 96) + 1
    If u < 1 Or u > Len(mPFB(p)) Then Exit Function
    Code = Asc(Mid$(mPFB(p), u, 1)) - 48
    If Code < 0 Or Code > 4 Then Code = 0
End Function

Private Function IsBusyCode(ByVal c As Long) As Boolean
    IsBusyCode = (c = 2 Or c = 3 Or (c = 1 And FT_TENTATIVE_BUSY))
End Function

' The worst moment for person p from quarter hour q0 up to q1 (busy beats
' tentative beats free); -1 if some of it isn't known.
Private Function WorstCode(ByVal p As Long, ByVal q0 As Long, ByVal q1 As Long) As Long
    Dim q As Long, c As Long, w As Long
    For q = q0 To q1 - 1
        c = Code(p, q)
        If c = -1 Then
            WorstCode = -1
            Exit Function
        End If
        If c = 4 Then c = 0                 ' working elsewhere is free
        If c > w Then w = c
    Next
    WorstCode = w
End Function

' The needed people whose calendars are read.
Private Function NeededPeople(need() As Long) As Long
    Dim p As Long, n As Long
    ReDim need(1 To mPN + 1)
    For p = 1 To mPN
        If mPRole(p) = "R" And mPState(p) = 1 Then
            n = n + 1
            need(n) = p
        End If
    Next
    NeededPeople = n
End Function

' Reads the calendars not read yet, one at a time. The list is worked out
' again as each one comes in, so times show from the start and firm up.
' People added meanwhile are read too.
Private Sub LoadCalendars()
    Dim p As Long, i As Long, gen As Long, fb As String, state As Long, done As Long, t0 As Single
    Dim addr As String
    If mLoading Then Exit Sub
    mLoading = True
    t0 = Timer
    On Error GoTo Quit
    Do
        p = 0
        mWaiting = 0
        For i = 1 To mPN
            If mPState(i) = 0 Then
                mWaiting = mWaiting + 1
                If p = 0 Then p = i
            End If
        Next
        If p = 0 Then Exit Do
        ShowHeader
        Me.Repaint
        gen = mGen
        addr = mPAddr(p)
        fb = FetchFB(addr, state)
        If gen = mGen Then
            p = PersonIndex(addr)
            If p > 0 Then
                mPFB(p) = fb
                mPFBDay(p) = Date
                mPState(p) = state
                mFBCache(LCase$(addr)) = Array(fb, state, Timer, CDbl(Date))
                done = done + 1
                BuildRows
            End If
        End If
        DoEvents                            ' keys typed meanwhile are dealt with here
    Loop
Quit:
    mLoading = False
    mWaiting = 0
    If done > 0 Then NVS_Log "Find a Time: " & done & " calendars read in " & Format$(Elapsed(t0), "0.0") & " s"
    BuildRows
End Sub

' Someone's free/busy from Outlook: a character for every 15 minutes, from
' midnight today for about a month. state = 1 if it was read, 2 if it can't be.
Private Function FetchFB(ByVal addr As String, ByRef state As Long) As String
    Dim r As Outlook.Recipient, s As String
    state = 2
    On Error Resume Next
    Set r = Application.Session.CreateRecipient(addr)
    If r Is Nothing Then Exit Function
    r.Resolve
    If Not r.Resolved Then Exit Function
    Err.Clear
    s = r.FreeBusy(Date, 15, True)
    If Err.Number = 0 And Len(s) > 0 Then
        FetchFB = s
        state = 1
    End If
End Function

' F5: reads everyone's calendar again.
Private Sub Reload()
    Dim i As Long
    For i = 1 To mPN
        If mPState(i) <> 3 Then
            mPState(i) = 0
            If mFBCache.Exists(LCase$(mPAddr(i))) Then mFBCache.Remove LCase$(mPAddr(i))
        End If
    Next
    BuildRows
    LoadCalendars
End Sub

'==============================================================================
' The times list
'==============================================================================

' Works out the rows of the times list and shows them, keeping the same
' time selected if it's still there.
Private Sub BuildRows()
    Dim keepT As Date, had As Boolean, keepQ As Long, r As Long
    If Not mBuilt Then Exit Sub
    On Error GoTo Failed
    If mRSel >= 1 And mRSel <= mRN Then
        If Selectable(mRSel) Then
            keepT = QTime(mRAt(mRSel))
            had = True
        End If
    End If
    mBase = Date
    mNowQ = QOf(Now, True)
    mLenQ = (mLen + 14) \ 15
    If mLenQ < 1 Then mLenQ = 1
    mRN = 0
    ReDim mRKind(1 To 32)
    ReDim mRA(1 To 32)
    ReDim mRB(1 To 32)
    ReDim mRAt(1 To 32)
    ReDim mRWho(1 To 32)
    ReDim mRBand(1 To 32)
    mNote = ""
    If mWinN > 0 Then TheirRows Else FreeRows
    SetBands

    mRSel = 0
    If had Then
        keepQ = QOf(keepT, False)
        For r = 1 To mRN
            If Selectable(r) And keepQ >= mRA(r) And keepQ < mRB(r) Then
                mRSel = r
                If keepQ <= LastStart(r) Then mRAt(r) = keepQ
                Exit For
            End If
        Next
    End If
    If mRSel = 0 Then mRSel = FirstGood()
Draw:
    On Error Resume Next                    ' drawing never stops with an error
    KeepRowVisible
    ShowHeader
    PaintTimes
    PaintPeople
    ShowAction
    Exit Sub
Failed:
    ' Shown in the list rather than as a VBA error.
    mRN = 0
    mRSel = 0
    mNote = "Couldn't work out the times: " & Err.Description
    ReDim mRKind(1 To 1)
    ReDim mRA(1 To 1)
    ReDim mRB(1 To 1)
    ReDim mRAt(1 To 1)
    ReDim mRWho(1 To 1)
    ReDim mRBand(1 To 1)
    mRN = 1
    mRKind(1) = K_NOTE
    Resume Draw
End Sub

' Everyone's free times: on each day, the stretches when every needed person
' is free for at least the meeting's length. A day with none gets the two
' best stretches that suit all but one person.
Private Sub FreeRows()
    Dim d As Long, q0 As Long, q1 As Long, need() As Long, nNeed As Long, n As Long, i As Long
    Dim rs() As Long, re() As Long
    nNeed = NeededPeople(need)
    For d = 1 To mDayN
        q0 = DayQ(d) + (mHourFrom + 14) \ 15
        q1 = DayQ(d) + mHourTo \ 15
        If q0 < mNowQ Then q0 = mNowQ
        If q1 - q0 >= mLenQ Then
            n = FreeRuns(q0, q1, need, nNeed, 0, rs, re)
            For i = 1 To n
                AddRow K_FREE, rs(i), re(i), ""
            Next
            If n = 0 Then
                AddRow K_NONE, q0, q0, ""
                AllButRows q0, q1, need, nNeed
            End If
        End If
    Next
    If mRN = 0 Then
        mNote = "No " & LengthText(mLen) & " times left in the hours looked in (" & HoursText() & ")."
        AddRow K_NOTE, mNowQ, mNowQ, ""
    End If
End Sub

' The stretches from quarter hour q0 up to q1 when all the needed people
' (but person skip) are free, at least the meeting's length: starts in rs,
' ends in re. Returns how many.
Private Function FreeRuns(ByVal q0 As Long, ByVal q1 As Long, need() As Long, ByVal nNeed As Long, _
                          ByVal skip As Long, rs() As Long, re() As Long) As Long
    Dim q As Long, a As Long, n As Long, i As Long, isFree As Boolean
    ReDim rs(1 To 16)
    ReDim re(1 To 16)
    a = -1
    For q = q0 To q1
        isFree = (q < q1)
        If isFree Then
            For i = 1 To nNeed
                If need(i) <> skip Then
                    If IsBusyCode(Code(need(i), q)) Then
                        isFree = False
                        Exit For
                    End If
                End If
            Next
        End If
        If isFree Then
            If a < 0 Then a = q
        ElseIf a >= 0 Then
            If q - a >= mLenQ Then
                n = n + 1
                If n > UBound(rs) Then
                    ReDim Preserve rs(1 To 2 * n)
                    ReDim Preserve re(1 To 2 * n)
                End If
                rs(n) = a
                re(n) = q
            End If
            a = -1
        End If
    Next
    FreeRuns = n
End Function

' A day when no time suits everyone: the two longest stretches that suit
' all but one person (the sooner one if they're as long), in time order.
Private Sub AllButRows(ByVal q0 As Long, ByVal q1 As Long, need() As Long, ByVal nNeed As Long)
    Dim i As Long, j As Long, n As Long, rs() As Long, re() As Long, w As Long, k As Long
    Dim cs(1 To 2) As Long, ce(1 To 2) As Long, cp(1 To 2) As Long, nc As Long
    If nNeed < 2 Then Exit Sub
    For i = 1 To nNeed
        n = FreeRuns(q0, q1, need, nNeed, need(i), rs, re)
        For j = 1 To n
            If nc < 2 Then
                nc = nc + 1
                w = nc
            Else
                If Longer(cs(1), ce(1), cs(2), ce(2)) Then w = 2 Else w = 1     ' the weaker of the two
                If Not Longer(rs(j), re(j), cs(w), ce(w)) Then w = 0
            End If
            If w > 0 Then
                cs(w) = rs(j)
                ce(w) = re(j)
                cp(w) = need(i)
            End If
        Next
    Next
    If nc = 2 Then
        If cs(2) < cs(1) Then
            k = cs(1): cs(1) = cs(2): cs(2) = k
            k = ce(1): ce(1) = ce(2): ce(2) = k
            k = cp(1): cp(1) = cp(2): cp(2) = k
        End If
    End If
    For k = 1 To nc
        AddRow K_ALLBUT, cs(k), ce(k), WhoLeftOut(cp(k), cs(k), ce(k))
    Next
End Sub

' True if the stretch a1-b1 is longer than a2-b2 (or as long and sooner).
Private Function Longer(ByVal a1 As Long, ByVal b1 As Long, ByVal a2 As Long, ByVal b2 As Long) As Boolean
    Longer = (b1 - a1 > b2 - a2) Or (b1 - a1 = b2 - a2 And a1 < a2)
End Function

' "Jane", or "Jane (away)" if she's out of office then.
Private Function WhoLeftOut(ByVal p As Long, ByVal a As Long, ByVal b As Long) As String
    Dim q As Long
    WhoLeftOut = ShortName(p)
    For q = a To b - 1
        If Code(p, q) = 3 Then
            WhoLeftOut = WhoLeftOut & " (away)"
            Exit Function
        End If
    Next
End Function

' Their times, in order, each split where who's busy changes: "works", or
' who can't make it.
Private Sub TheirRows()
    Dim i As Long, j As Long, m As Long, ws() As Date, we() As Date, t As Date
    Dim qa As Long, qb As Long, q As Long, e As Long, key As String, need() As Long, nNeed As Long

    nNeed = NeededPeople(need)
    ReDim ws(1 To mWinN)
    ReDim we(1 To mWinN)
    For i = 1 To mWinN                      ' in time order ...
        t = mWinS(i)
        j = m
        Do While j >= 1
            If ws(j) <= t Then Exit Do
            ws(j + 1) = ws(j)
            we(j + 1) = we(j)
            j = j - 1
        Loop
        ws(j + 1) = t
        we(j + 1) = mWinE(i)
        m = m + 1
    Next
    j = 0                                   ' ... and overlapping ones joined
    For i = 1 To m
        If j > 0 Then
            If ws(i) <= we(j) Then
                If we(i) > we(j) Then we(j) = we(i)
            Else
                j = j + 1
                ws(j) = ws(i)
                we(j) = we(i)
            End If
        Else
            j = 1
            ws(1) = ws(i)
            we(1) = we(i)
        End If
    Next
    m = j

    For i = 1 To m
        qa = QOf(ws(i), True)
        qb = QOf(we(i), False)
        If qb > qa Then
            If qb <= mNowQ Then
                AddRow K_PAST, qa, qb, ""
            Else
                If qa < mNowQ Then qa = mNowQ
                q = qa
                Do While q < qb
                    key = BusyKey(q, need, nNeed)
                    e = q + 1
                    Do While e < qb
                        If BusyKey(e, need, nNeed) <> key Then Exit Do
                        e = e + 1
                    Loop
                    If key = "" Then
                        If e - q >= mLenQ Then AddRow K_WORKS, q, e, "" Else AddRow K_SHORT, q, e, ""
                    ElseIf key = "?" Then
                        AddRow K_FAR, q, e, ""
                    Else
                        AddRow K_BUSY, q, e, KeyNames(key)
                    End If
                    q = e
                Loop
            End If
        End If
    Next
    If mRN = 0 Then
        mNote = "Those times are too short to check."
        AddRow K_NOTE, mNowQ, mNowQ, ""
    End If
End Sub

' Who's busy in quarter hour q, as "3,5," (people's numbers); "?" if a
' calendar isn't known then.
Private Function BusyKey(ByVal q As Long, need() As Long, ByVal nNeed As Long) As String
    Dim i As Long, c As Long
    For i = 1 To nNeed
        c = Code(need(i), q)
        If c = -1 Then
            BusyKey = "?"
            Exit Function
        End If
        If IsBusyCode(c) Then BusyKey = BusyKey & need(i) & ","
    Next
End Function

Private Function KeyNames(ByVal key As String) As String
    Dim v As Variant
    For Each v In Split(key, ",")
        If CStr(v) <> "" Then
            If KeyNames <> "" Then KeyNames = KeyNames & ", "
            KeyNames = KeyNames & ShortName(CLng(v))
        End If
    Next
End Function

Private Sub AddRow(ByVal kind As Long, ByVal a As Long, ByVal b As Long, ByVal who As String)
    Dim n As Long
    mRN = mRN + 1
    If mRN > UBound(mRKind) Then
        n = 2 * mRN
        ReDim Preserve mRKind(1 To n)
        ReDim Preserve mRA(1 To n)
        ReDim Preserve mRB(1 To n)
        ReDim Preserve mRAt(1 To n)
        ReDim Preserve mRWho(1 To n)
        ReDim Preserve mRBand(1 To n)
    End If
    mRKind(mRN) = kind
    mRA(mRN) = a
    mRB(mRN) = b
    mRWho(mRN) = who
    mRAt(mRN) = a
    If kind = K_FREE Or kind = K_ALLBUT Or kind = K_WORKS Then
        ' On the hour or half hour, if the meeting still fits.
        If a Mod 2 <> 0 And a + 1 + mLenQ <= b Then mRAt(mRN) = a + 1
    End If
End Sub

' Every other day is shaded, so the days stand apart.
Private Sub SetBands()
    Dim r As Long, band As Boolean
    For r = 1 To mRN
        If r > 1 Then
            If DayOfQ(mRA(r)) <> DayOfQ(mRA(r - 1)) Then band = Not band
        End If
        mRBand(r) = band
    Next
End Sub

' Rows that are times (not notes): they can be picked and invited for.
Private Function Selectable(ByVal r As Long) As Boolean
    If r < 1 Or r > mRN Then Exit Function
    Select Case mRKind(r)
        Case K_FREE, K_ALLBUT, K_WORKS, K_BUSY, K_SHORT, K_FAR: Selectable = True
    End Select
End Function

' The first time that suits everyone (or, if none does, the first time listed).
Private Function FirstGood() As Long
    Dim r As Long
    For r = 1 To mRN
        If mRKind(r) = K_FREE Or mRKind(r) = K_WORKS Then
            FirstGood = r
            Exit Function
        End If
    Next
    FirstGood = NextSelectable(0, 1)
End Function

' The latest the meeting can start in row r.
Private Function LastStart(ByVal r As Long) As Long
    LastStart = mRB(r) - mLenQ
    If LastStart < mRA(r) Then LastStart = mRA(r)
End Function

' The next row that can be picked after (stepDir = 1) or before (-1) row r;
' 0 if there isn't one.
Private Function NextSelectable(ByVal r As Long, ByVal stepDir As Long) As Long
    r = r + stepDir
    Do While r >= 1 And r <= mRN
        If Selectable(r) Then
            NextSelectable = r
            Exit Function
        End If
        r = r + stepDir
    Loop
End Function

' People busy during a meeting starting at quarter hour q: role "R" for the
' needed ones, "O" for the optional ones.
Private Function BusyDuring(ByVal q As Long, ByVal role As String) As String
    Dim p As Long
    For p = 1 To mPN
        If mPRole(p) = role And mPState(p) = 1 Then
            If IsBusyCode(WorstCode(p, q, q + mLenQ)) Then
                If BusyDuring <> "" Then BusyDuring = BusyDuring & ", "
                BusyDuring = BusyDuring & ShortName(p)
            End If
        End If
    Next
End Function

' The list's title, and the calendars it's still waiting for.
Private Sub ShowHeader()
    Dim s As String, p As Long, n As Long, x As Single
    On Error Resume Next
    If mWinN > 0 Then
        s = "Their times: who's free"
    Else
        For p = 2 To mPN
            If mPRole(p) = "R" And mPState(p) <> 3 Then n = n + 1
        Next
        If n = 0 Then s = "When you're free" Else s = "When everyone's free"
    End If
    If lblTimes.Caption <> s Then lblTimes.Caption = s
    If mWaiting > 0 Then
        s = "checking " & mWaiting & IIf(mWaiting = 1, " calendar...", " calendars...")
    ElseIf mWinN > 0 Then
        s = ""
    Else
        s = HoursText()
    End If
    If lblTimesInfo.Caption <> s Then lblTimesInfo.Caption = s
    x = lblTimes.Left + lblTimes.Width + 8
    lblTimesInfo.Move x, lblTimes.Top + 1, lblLen.Left - 4 - x, 11
End Sub

' The status line: what Enter does.
Private Sub ShowAction()
    Dim r As Long, s As String, st As Date, busy As String
    r = mRSel
    If Selectable(r) Then
        st = QTime(mRAt(r))
        s = "Enter: invite for " & Format$(st, "ddd m/d") & ", " & RangeText(st, st + mLen / 1440#)
        busy = BusyDuring(mRAt(r), "R")
        If mRKind(r) = K_FAR Then
            s = s & "  - calendars can't be seen that far ahead"
        ElseIf busy <> "" Then
            s = s & "  - " & busy & " busy then"
        Else
            busy = BusyDuring(mRAt(r), "O")
            If busy <> "" Then s = s & "  - " & busy & " (optional) busy then"
        End If
    ElseIf mNote <> "" Then
        s = mNote
    End If
    SetStatus s
End Sub

' Enter: the invite for the selected time.
Private Sub InviteSelected()
    If Not Selectable(mRSel) Then
        Beep
        Exit Sub
    End If
    CreateInvite QTime(mRAt(mRSel))
End Sub

' Opens Outlook's meeting invite at st with everyone on it: needed people
' required, optional people optional, and the people outside the firm too
' if that's ticked. Nothing is sent.
Private Sub CreateInvite(ByVal st As Date)
    Dim ap As Outlook.AppointmentItem, rc As Outlook.Recipient, p As Long, n As Long
    On Error GoTo Fail
    Set ap = Application.CreateItem(olAppointmentItem)
    ap.MeetingStatus = olMeeting
    ap.Subject = mTopic
    ap.Start = st
    ap.Duration = mLen
    For p = 2 To mPN
        Set rc = ap.Recipients.Add(mPAddr(p))
        If mPRole(p) = "O" Then rc.Type = olOptional Else rc.Type = olRequired
        n = n + 1
    Next
    If chkOut.Value = True Then
        For p = 1 To mON
            Set rc = ap.Recipients.Add(mOAddr(p))
            If mORole(p) = "O" Then rc.Type = olOptional Else rc.Type = olRequired
            n = n + 1
        Next
    End If
    ap.Recipients.ResolveAll
    NVS_Log "Find a Time: invite for " & Format$(st, "ddd m/d h:nn AM/PM") & " with " & n & " people"
    HideWindow
    ap.Display
    Exit Sub
Fail:
    MsgBox "Couldn't open the invite: " & Err.Description, vbExclamation + vbMsgBoxSetForeground, "Find a Time"
End Sub

' Ctrl+C: the free times as text for a reply, from the selected day on, a
' line a day:   Tue 11/24: 9 - 10:30 AM, 2 - 4 PM
Private Sub CopyTimes()
    Dim r As Long, first As Long, s As String, ln As String, nDays As Long, lastDay As Date
    Dim a As Long, b As Long, zone As String, d As Date
    On Error GoTo Failed
    first = mRSel
    If first < 1 Then first = 1
    Do While first > 1 And first <= mRN
        If DayOfQ(mRA(first - 1)) <> DayOfQ(mRA(first)) Then Exit Do
        first = first - 1
    Loop
    For r = first To mRN
        If mRKind(r) = K_FREE Or mRKind(r) = K_WORKS Then
            d = DayOfQ(mRA(r))
            If d <> lastDay Then
                If nDays = FT_COPY_DAYS Then Exit For
                If ln <> "" Then s = s & ln & vbCrLf
                ln = Format$(d, "ddd m/d") & ": "
                nDays = nDays + 1
                lastDay = d
            Else
                ln = ln & ", "
            End If
            ' On the half hour, if the meeting still fits.
            a = mRA(r) + (mRA(r) Mod 2)
            b = mRB(r) - (mRB(r) Mod 2)
            If b - a < mLenQ Then
                a = mRA(r)
                b = mRB(r)
            End If
            ln = ln & RangeText(QTime(a), QTime(b))
        End If
    Next
    If ln <> "" Then s = s & ln
    If s = "" Then
        Beep
        If mWinN > 0 Then SetStatus "None of their times work, so there's nothing to copy." Else SetStatus "No free times to copy."
        Exit Sub
    End If
    zone = ZoneShort()
    If zone <> "" Then s = s & vbCrLf & "(all times " & zone & ")"
    If PutOnClipboard(s) Then
        SetStatus "Copied the free times for " & nDays & IIf(nDays = 1, " day", " days") & _
                  " - paste them into your reply (Ctrl+V)."
    Else
        Beep
        SetStatus "Couldn't copy: another program is using the clipboard. Try again."
    End If
    Exit Sub
Failed:
    SetStatus "Couldn't copy the times: " & Err.Description
End Sub

' Puts text on the clipboard.
Private Function PutOnClipboard(ByVal s As String) As Boolean
    Dim h As LongPtr, p As LongPtr, owner As LongPtr, i As Long, opened As Boolean
    h = GlobalAlloc(&H42, (Len(s) + 1) * 2)        ' movable and zero-filled: the zero ends the text
    If h = 0 Then Exit Function
    p = GlobalLock(h)
    If p = 0 Then
        GlobalFree h
        Exit Function
    End If
    If Len(s) > 0 Then CopyBytes p, StrPtr(s), Len(s) * 2
    GlobalUnlock h
    owner = mHwnd
    If owner = 0 Then owner = GetForegroundWindow()
    For i = 1 To 10                                ' another program may have it open for a moment
        If OpenClipboard(owner) <> 0 Then
            opened = True
            Exit For
        End If
        DoEvents
    Next
    If Not opened Then
        GlobalFree h
        Exit Function
    End If
    EmptyClipboard
    If SetClipboardData(13, h) = 0 Then GlobalFree h Else PutOnClipboard = True    ' 13: Unicode text
    CloseClipboard
End Function

' Your time zone the way emails write it ("ET"); "" if it isn't one of these.
Private Function ZoneShort() As String
    Dim zid As String
    On Error Resume Next
    zid = Application.TimeZones.CurrentTimeZone.ID
    On Error GoTo 0
    Select Case zid
        Case "Eastern Standard Time", "US Eastern Standard Time": ZoneShort = "ET"
        Case "Central Standard Time": ZoneShort = "CT"
        Case "Mountain Standard Time", "US Mountain Standard Time": ZoneShort = "MT"
        Case "Pacific Standard Time": ZoneShort = "PT"
        Case "GMT Standard Time": ZoneShort = "UK time"
        Case "W. Europe Standard Time", "Romance Standard Time", "Central Europe Standard Time": ZoneShort = "CET"
        Case "China Standard Time": ZoneShort = "HKT"
        Case "Singapore Standard Time": ZoneShort = "SGT"
        Case "Tokyo Standard Time": ZoneShort = "JST"
        Case "India Standard Time": ZoneShort = "IST"
    End Select
End Function

'------------------------------------------------------------------------------
' Words for times
'------------------------------------------------------------------------------

' "Today", "Tomorrow" or "Wed 11/25".
Private Function DayText(ByVal d As Date) As String
    If d = Date Then
        DayText = "Today"
    ElseIf d = Date + 1 Then
        DayText = "Tomorrow"
    Else
        DayText = Format$(d, "ddd m/d")
    End If
End Function

' "2 - 4 PM", "11 AM - 12:30 PM".
Private Function RangeText(ByVal t1 As Date, ByVal t2 As Date) As String
    If (Hour(t1) < 12) = (Hour(t2) < 12) And Int(t1) = Int(t2) Then
        RangeText = BareClock(t1) & " - " & ClockText(t2)
    Else
        RangeText = ClockText(t1) & " - " & ClockText(t2)
    End If
End Function

' "2" or "2:30": a time without its AM / PM.
Private Function BareClock(ByVal t As Date) As String
    BareClock = ClockText(t)
    BareClock = Left$(BareClock, Len(BareClock) - 3)
End Function

' The hours looked in, as "8 AM - 6 PM".
Private Function HoursText() As String
    HoursText = RangeText(Date + mHourFrom / 1440#, Date + mHourTo / 1440#)
End Function

Private Function LengthText(ByVal mins As Long) As String
    Select Case mins
        Case 60: LengthText = "1 hour"
        Case 120: LengthText = "2 hours"
        Case Else: LengthText = mins & " min"
    End Select
End Function

' Minutes for what's in the Length box ("45", "45 min", "1 hour", "1.5h"),
' 0 if it isn't a length.
Private Function ParseLength(ByVal s As String) As Long
    Dim v As Double
    s = LCase$(Trim$(s))
    v = Val(s)
    If InStr(s, "h") > 0 Then v = v * 60
    If v >= 5 And v <= 600 Then ParseLength = CLng(v)
End Function

' The people: name, "optional", and what they're doing at the selected time.
Private Sub PaintPeople()
    Dim k As Long, i As Long, rowW As Single, back As Long, col As Long, tagW As Single
    If mPSlotsMade = 0 Then Exit Sub
    On Error Resume Next
    mPainting = True
    sbPeople.Visible = (mPN > mPSlots)
    If sbPeople.Visible Then
        sbPeople.Min = 1
        sbPeople.Max = mPN - mPSlots + 1
        sbPeople.LargeChange = IIf(mPSlots > 1, mPSlots - 1, 1)
        sbPeople.SmallChange = 1
        If sbPeople.Value <> mPTop Then sbPeople.Value = mPTop
        rowW = sbPeople.Left
    Else
        rowW = fraPeople.InsideWidth
    End If
    mPainting = False
    lblPeopleHit.Width = rowW
    For k = 0 To mPSlots - 1
        i = mPTop + k
        If i <= mPN Then
            If i = mPSel And mFocus = 1 Then
                back = C_SEL
            ElseIf i = mPHover Then
                back = C_HOVER
            Else
                back = C_WHITE
            End If
            mPRowBack(k).Width = rowW
            If mPRowBack(k).BackColor <> back Then mPRowBack(k).BackColor = back
            mPRowBack(k).Visible = True
            mPRowStat(k).Caption = PersonWord(i, col)
            mPRowStat(k).ForeColor = col
            mPRowStat(k).Left = rowW - 60
            mPRowStat(k).Visible = True
            If mPRole(i) = "O" Then tagW = 42 Else tagW = 0
            mPRowTag(k).Left = rowW - 60 - tagW
            mPRowTag(k).Visible = (tagW > 0)
            mPRowName(k).Width = rowW - 66 - tagW
            mPRowName(k).Caption = mPName(i)
            mPRowName(k).Visible = True
        Else
            mPRowBack(k).Visible = False
            mPRowName(k).Visible = False
            mPRowTag(k).Visible = False
            mPRowStat(k).Visible = False
        End If
    Next
End Sub

' What person i is doing at the selected time, and its color.
Private Function PersonWord(ByVal i As Long, ByRef col As Long) As String
    col = C_GRAY
    Select Case mPState(i)
        Case 0
            PersonWord = "checking..."
        Case 2, 3
            PersonWord = "no calendar"
        Case Else
            If Not Selectable(mRSel) Then Exit Function
            Select Case WorstCode(i, mRAt(mRSel), mRAt(mRSel) + mLenQ)
                Case -1: PersonWord = "?"
                Case 0: PersonWord = "free": col = C_GREEN
                Case 1: PersonWord = "tentative": col = C_AMBER
                Case 2: PersonWord = "busy": col = C_RED
                Case 3: PersonWord = "away": col = C_PURPLE
            End Select
    End Select
End Function

' The times: the day (on its first row), the time, and who's busy.
Private Sub PaintTimes()
    Dim k As Long, r As Long, rowW As Single, back As Long, d As Date, newDay As Boolean
    Dim tCap As String, tCol As Long, nCap As String, nCol As Long, wide As Boolean
    If mTSlotsMade = 0 Then Exit Sub
    On Error Resume Next
    mPainting = True
    sbTimes.Visible = (mRN > mTSlots)
    If sbTimes.Visible Then
        sbTimes.Min = 1
        sbTimes.Max = mRN - mTSlots + 1
        sbTimes.LargeChange = IIf(mTSlots > 1, mTSlots - 1, 1)
        sbTimes.SmallChange = 1
        If sbTimes.Value <> mRTop Then sbTimes.Value = mRTop
        rowW = sbTimes.Left
    Else
        rowW = fraTimes.InsideWidth
    End If
    mPainting = False
    lblTimesHit.Width = rowW
    For k = 0 To mTSlots - 1
        r = mRTop + k
        If r <= mRN Then
            If r = mRSel Then
                If mFocus = 2 Then back = C_SEL Else back = C_SEL_OFF
            ElseIf r = mTHover And Selectable(r) Then
                back = C_HOVER
            ElseIf mRBand(r) Then
                back = C_BAND
            Else
                back = C_WHITE
            End If
            mTRowBack(k).Width = rowW
            If mTRowBack(k).BackColor <> back Then mTRowBack(k).BackColor = back
            mTRowBack(k).Visible = True

            d = DayOfQ(mRA(r))
            newDay = (k = 0)
            If Not newDay Then newDay = (DayOfQ(mRA(r - 1)) <> d)
            If newDay And mRKind(r) <> K_NOTE Then mTRowDay(k).Caption = DayText(d) Else mTRowDay(k).Caption = ""
            mTRowDay(k).Font.Bold = True
            mTRowDay(k).Visible = True

            RowWords r, tCap, tCol, nCap, nCol, wide
            mTRowTime(k).Caption = tCap
            mTRowTime(k).ForeColor = tCol
            If wide Then mTRowTime(k).Width = rowW - 80 Else mTRowTime(k).Width = 118
            mTRowTime(k).Visible = True
            mTRowNote(k).Caption = nCap
            mTRowNote(k).ForeColor = nCol
            mTRowNote(k).Width = rowW - 200
            mTRowNote(k).Visible = Not wide
        Else
            mTRowBack(k).Visible = False
            mTRowDay(k).Visible = False
            mTRowTime(k).Visible = False
            mTRowNote(k).Visible = False
        End If
    Next
End Sub

' The words of row r: its time (tCap, in color tCol) and a note after it
' (nCap, nCol). wide = the time's words take the whole row.
Private Sub RowWords(ByVal r As Long, ByRef tCap As String, ByRef tCol As Long, _
                     ByRef nCap As String, ByRef nCol As Long, ByRef wide As Boolean)
    Dim opt As String
    tCap = RangeText(QTime(mRA(r)), QTime(mRB(r)))
    tCol = C_TEXT
    nCap = ""
    nCol = C_GRAY
    wide = False
    Select Case mRKind(r)
        Case K_FREE
            opt = BusyDuring(mRAt(r), "O")
            If opt <> "" Then
                nCap = opt & " busy (optional)"
                nCol = C_AMBER
            End If
        Case K_ALLBUT
            nCap = "all but " & mRWho(r)
            nCol = C_AMBER
        Case K_NONE
            tCap = "no time with everyone free"
            tCol = C_GRAY
            wide = True
        Case K_WORKS
            opt = BusyDuring(mRAt(r), "O")
            If opt <> "" Then
                nCap = "works, but " & opt & " (optional) busy"
                nCol = C_AMBER
            Else
                nCap = "works"
                nCol = C_GREEN
            End If
        Case K_BUSY
            nCap = mRWho(r) & " busy"
            nCol = C_RED
        Case K_SHORT
            nCap = "everyone free, but only for " & (mRB(r) - mRA(r)) * 15 & " min"
            nCol = C_AMBER
        Case K_PAST
            tCol = C_GRAY
            nCap = "already past"
        Case K_FAR
            nCap = "too far ahead to see calendars"
        Case K_NOTE
            tCap = mNote
            tCol = C_GRAY
            wide = True
    End Select
End Sub

' Puts the keyboard in a list: 1 the people, 2 the times.
Private Sub FocusList(ByVal which As Long)
    On Error Resume Next
    HideSug
    If which = 1 Then
        If mPSel < 1 Then mPSel = 1
        txtPeopleKeys.SetFocus
    Else
        txtTimesKeys.SetFocus
    End If
    SetFocusMark which
End Sub

' Which list has the keyboard, so its selection is drawn as in use.
Private Sub SetFocusMark(ByVal which As Long)
    If mFocus = which Then Exit Sub
    mFocus = which
    PaintPeople
    PaintTimes
End Sub

Private Sub SelectRow(ByVal r As Long)
    If Not Selectable(r) Then Exit Sub
    mRSel = r
    KeepRowVisible
    PaintTimes
    PaintPeople
    ShowAction
End Sub

Private Sub MoveRow(ByVal stepDir As Long)
    Dim r As Long
    r = NextSelectable(mRSel, stepDir)
    If r > 0 Then
        SelectRow r
    ElseIf stepDir < 0 And mRTop > 1 Then
        mRTop = 1                           ' show the note at the top
        PaintTimes
    End If
End Sub

' Page Down: the first time of the next day. Page Up: the first time of this
' day, or of the day before if already there.
Private Sub MoveDay(ByVal stepDir As Long)
    Dim r As Long, d As Date, f As Long
    If Not Selectable(mRSel) Then
        SelectRow NextSelectable(0, 1)
        Exit Sub
    End If
    d = DayOfQ(mRA(mRSel))
    If stepDir > 0 Then
        r = mRSel
        Do
            r = NextSelectable(r, 1)
            If r = 0 Then Exit Sub
            If DayOfQ(mRA(r)) <> d Then Exit Do
        Loop
        SelectRow r
    Else
        f = FirstOfDay(mRSel)
        If f <> mRSel Then
            SelectRow f
        Else
            r = NextSelectable(f, -1)
            If r > 0 Then SelectRow FirstOfDay(r)
        End If
    End If
End Sub

' The first time listed on row r's day.
Private Function FirstOfDay(ByVal r As Long) As Long
    Dim d As Date, k As Long
    d = DayOfQ(mRA(r))
    FirstOfDay = r
    k = NextSelectable(r, -1)
    Do While k > 0
        If DayOfQ(mRA(k)) <> d Then Exit Do
        FirstOfDay = k
        k = NextSelectable(k, -1)
    Loop
End Function

' Left / Right: the meeting starts 15 minutes earlier / later, inside the
' selected time.
Private Sub Nudge(ByVal stepDir As Long)
    Dim q As Long
    If Not Selectable(mRSel) Then Exit Sub
    q = mRAt(mRSel) + stepDir
    If q < mRA(mRSel) Or q > LastStart(mRSel) Then Exit Sub
    mRAt(mRSel) = q
    PaintTimes
    PaintPeople
    ShowAction
End Sub

Private Sub KeepRowVisible()
    Dim want As Long
    If mTSlots < 1 Then Exit Sub
    If mRSel >= 1 And mRSel <= mRN Then
        want = mRSel
        If want > 1 Then
            If mRKind(want - 1) = K_NONE Then want = want - 1     ' and the note above it
        End If
        If want < mRTop Then mRTop = want
        If mRSel > mRTop + mTSlots - 1 Then mRTop = mRSel - mTSlots + 1
    End If
    If mRTop > mRN - mTSlots + 1 Then mRTop = mRN - mTSlots + 1
    If mRTop < 1 Then mRTop = 1
End Sub

Private Sub ScrollTimes(ByVal delta As Long)
    Dim t As Long
    t = mRTop + delta
    If t > mRN - mTSlots + 1 Then t = mRN - mTSlots + 1
    If t < 1 Then t = 1
    If t = mRTop Then Exit Sub
    mRTop = t
    PaintTimes
End Sub

Private Sub SelectPerson(ByVal i As Long)
    If i > mPN Then i = mPN
    If i < 1 Then i = 1
    mPSel = i
    KeepPersonVisible
    PaintPeople
End Sub

Private Sub KeepPersonVisible()
    If mPSlots < 1 Then Exit Sub
    If mPSel >= 1 Then
        If mPSel < mPTop Then mPTop = mPSel
        If mPSel > mPTop + mPSlots - 1 Then mPTop = mPSel - mPSlots + 1
    End If
    If mPTop > mPN - mPSlots + 1 Then mPTop = mPN - mPSlots + 1
    If mPTop < 1 Then mPTop = 1
End Sub

Private Sub ScrollPeople(ByVal delta As Long)
    Dim t As Long
    t = mPTop + delta
    If t > mPN - mPSlots + 1 Then t = mPN - mPSlots + 1
    If t < 1 Then t = 1
    If t = mPTop Then Exit Sub
    mPTop = t
    PaintPeople
End Sub

' Keys that work wherever you are in the window. True if it was one of them.
Private Function CommonKey(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer) As Boolean
    Dim k As Long
    k = KeyCode
    CommonKey = True
    Select Case k
        Case vbKeyEscape
            If fraSug.Visible Then HideSug Else HideWindow
        Case vbKeyF5
            Reload
        Case vbKeyReturn
            If Shift = 2 Then InviteSelected Else CommonKey = False      ' Ctrl+Enter: the invite
        Case vbKeyK
            If Shift = 2 Then PickFromAddressBook Else CommonKey = False ' Ctrl+K: address book
        Case vbKeyA, vbKeyP, vbKeyT, vbKeyW, vbKeyL
            If Shift = 4 Then GoAlt k Else CommonKey = False             ' in case a label's Alt key didn't
        Case Else
            CommonKey = False
    End Select
    If CommonKey Then KeyCode = 0
End Function

Private Sub GoAlt(ByVal key As Long)
    On Error Resume Next
    Select Case key
        Case vbKeyA
            txtAdd.SetFocus
            SetFocusMark 0
        Case vbKeyP
            FocusList 1
        Case vbKeyT
            txtTheir.SetFocus
            SetFocusMark 0
        Case vbKeyW
            FocusList 2
        Case vbKeyL
            cboLen.SetFocus
            SetFocusMark 0
    End Select
End Sub

' A letter typed in a list starts adding a person.
Private Sub TypeToAdd(ByVal KeyAscii As MSForms.ReturnInteger)
    Dim ch As String
    If KeyAscii < 33 Then
        KeyAscii = 0
        Exit Sub
    End If
    ch = Chr$(KeyAscii)
    KeyAscii = 0
    txtAdd.SetFocus
    SetFocusMark 0
    txtAdd.Text = txtAdd.Text & ch
    txtAdd.SelStart = Len(txtAdd.Text)
End Sub

' Up / Down in Length: the next shorter / longer of 15, 30, 45 minutes,
' 1 hour, 90 minutes, 2 hours.
Private Sub StepLength(ByVal stepDir As Long)
    Dim v As Variant, i As Long, m As Long
    v = Array(15, 30, 45, 60, 90, 120)
    m = mLen
    If stepDir > 0 Then
        For i = 0 To UBound(v)
            If v(i) > mLen Then
                m = v(i)
                Exit For
            End If
        Next
    Else
        For i = UBound(v) To 0 Step -1
            If v(i) < mLen Then
                m = v(i)
                Exit For
            End If
        Next
    End If
    If m <> mLen Then cboLen.Text = LengthText(m)       ' cboLen_Change does the rest
End Sub

'------------------------------------------------------------------------------
' Reading typed times
'------------------------------------------------------------------------------

' Reads times typed the way people write them in emails - "tue 2-4pm, wed
' 10-11:30 ET", "thursday after 3", "11/26 9am-12", "tomorrow morning",
' "mon or tue 2-4" - into windows of your local time. Returns how many there
' are; info says how they were read (and anything that couldn't be).
Private Function ParseTimes(ByVal typed As String, wS() As Date, wE() As Date, ByRef info As String) As Long
    Dim s As String, pieces() As String, i As Long, j As Long, n As Long, k As Long, cnt As Long
    Dim pDays() As Collection, pHas() As Boolean, pS() As Long, pE() As Long, pZone() As String
    Dim pOK() As Boolean, zoneAll As String, useDays As Collection, d As Variant, z As String
    Dim st As Date, en As Date, bad As String, shown As String, converted As Boolean

    info = ""
    s = NormalizeTimes(typed)
    If Trim$(Replace(s, ",", "")) = "" Then Exit Function
    pieces = Split(s, ",")
    n = UBound(pieces)
    ReDim pDays(0 To n)
    ReDim pHas(0 To n)
    ReDim pS(0 To n)
    ReDim pE(0 To n)
    ReDim pZone(0 To n)
    ReDim pOK(0 To n)
    For i = 0 To n
        Set pDays(i) = New Collection
        If Trim$(pieces(i)) <> "" Then
            pOK(i) = ParsePiece(Trim$(pieces(i)), pDays(i), pHas(i), pS(i), pE(i), pZone(i))
            If Not pOK(i) Then bad = bad & ", '" & Trim$(pieces(i)) & "'"
            If pZone(i) <> "" And zoneAll = "" Then zoneAll = pZone(i)
        End If
    Next

    ' Days with no times take the times after them ("mon or tue 2-4"); times
    ' with no day take the days before them ("tue 10-11, 2-4").
    For i = 0 To n
        If pOK(i) And Not pHas(i) And pDays(i).Count > 0 Then
            For j = i + 1 To n
                If pOK(j) And pHas(j) Then
                    pS(i) = pS(j)
                    pE(i) = pE(j)
                    pHas(i) = True
                    Exit For
                End If
            Next
        End If
    Next

    cnt = 0
    ReDim wS(1 To 64)
    ReDim wE(1 To 64)
    Set useDays = Nothing
    For i = 0 To n
        If pOK(i) Then
            If pDays(i).Count > 0 Then Set useDays = pDays(i)
            If Not pHas(i) Then
                pS(i) = mHourFrom
                pE(i) = mHourTo
            End If
            z = pZone(i)
            If z = "" Then z = zoneAll
            If useDays Is Nothing Then
                ' No day at all: every day shown.
                For k = 1 To mDayN
                    AddWindow wS, wE, cnt, mDay(k), pS(i), pE(i), z, converted
                Next
            Else
                For Each d In useDays
                    AddWindow wS, wE, cnt, CDate(d), pS(i), pE(i), z, converted
                Next
            End If
        End If
    Next
    ParseTimes = cnt

    ' What was understood, in local time.
    For i = 1 To cnt
        If i <= 6 Then
            If shown <> "" Then shown = shown & ";  "
            shown = shown & Format$(wS(i), "ddd m/d ") & ClockText(wS(i)) & "-" & ClockText(wE(i))
        End If
    Next
    If cnt > 6 Then shown = shown & ";  and " & (cnt - 6) & " more"
    If converted Then shown = shown & "  (in your time)"
    If bad <> "" Then
        If shown <> "" Then shown = shown & "   "
        shown = shown & "Couldn't read " & Mid$(bad, 3)
    End If
    info = shown
End Function

Private Sub AddWindow(wS() As Date, wE() As Date, ByRef cnt As Long, ByVal dayDate As Date, _
                      ByVal sMin As Long, ByVal eMin As Long, ByVal zone As String, ByRef converted As Boolean)
    Dim st As Date, en As Date
    If eMin <= sMin Then Exit Sub
    If cnt >= UBound(wS) Then Exit Sub
    st = dayDate + sMin / 1440#
    en = dayDate + eMin / 1440#
    If zone <> "" Then
        st = ToLocal(st, zone, converted)
        en = ToLocal(en, zone, converted)
    End If
    cnt = cnt + 1
    wS(cnt) = st
    wE(cnt) = en
End Sub

' "2:00 PM", or "2 PM" on the hour.
Private Function ClockText(ByVal t As Date) As String
    If Minute(t) = 0 Then ClockText = Format$(t, "h AM/PM") Else ClockText = Format$(t, "h:nn AM/PM")
End Function

' Lower case, one kind of dash, "to" / "until" / "between ... and" as dashes,
' and commas between separate times ("or", "and", ";", new lines).
Private Function NormalizeTimes(ByVal s As String) As String
    Dim p As Long, q As Long
    s = " " & LCase$(s) & " "
    s = Replace(s, ChrW$(8211), "-")
    s = Replace(s, ChrW$(8212), "-")
    s = Replace(s, vbCr, ",")
    s = Replace(s, vbLf, ",")
    s = Replace(s, vbTab, " ")
    s = Replace(s, ";", ",")
    s = Replace(s, "a.m.", "am")
    s = Replace(s, "p.m.", "pm")
    s = Replace(s, "a.m", "am")
    s = Replace(s, "p.m", "pm")
    s = Replace(s, "afternoon", "#aft#")            ' so "noon" below doesn't hit it
    s = Replace(s, "noon", "12pm")
    s = Replace(s, "midday", "12pm")
    s = Replace(s, "#aft#", "afternoon")
    s = Replace(s, " to ", " - ")
    s = Replace(s, " till ", " - ")
    s = Replace(s, " until ", " - ")
    s = Replace(s, " thru ", " - ")
    s = Replace(s, " through ", " - ")
    ' "between 2 and 4" -> "2 - 4"
    p = InStr(s, " between ")
    Do While p > 0
        q = InStr(p + 9, s, " and ")
        If q > 0 Then s = Left$(s, q - 1) & " - " & Mid$(s, q + 5)
        s = Left$(s, p) & Mid$(s, p + 9)
        p = InStr(s, " between ")
    Loop
    s = Replace(s, " or ", ",")
    s = Replace(s, " and ", ",")
    s = Replace(s, " & ", ",")
    NormalizeTimes = s
End Function

' One part of what was typed: its days (if any) and its times (if any).
' Words it doesn't know are skipped. False if it has neither days nor times.
Private Function ParsePiece(ByVal pc As String, ByVal days As Collection, ByRef hasTime As Boolean, _
                            ByRef sMin As Long, ByRef eMin As Long, ByRef zone As String) As Boolean
    Dim toks() As String, i As Long, t As String, wd As Long, nxt As Boolean, dt As Date
    Dim tm(1 To 4) As Long, mer(1 To 4) As Long, nT As Long, aft As Boolean, bef As Boolean
    Dim part As String, m As Long, mr As Long, mo As Long, z As String, dy As Long

    hasTime = False
    zone = ""
    pc = Replace(pc, "-", " - ")
    Do While InStr(pc, "  ") > 0
        pc = Replace(pc, "  ", " ")
    Loop
    toks = Split(Trim$(pc), " ")
    i = 0
    Do While i <= UBound(toks)
        t = toks(i)
        z = ZoneID(t)
        If t = "" Or t = "-" Then
            ' a dash just joins two times
        ElseIf t = "next" Then
            nxt = True
        ElseIf DayWord(t, wd) Then
            days.Add NextDay(wd, nxt)
            nxt = False
        ElseIf DayList(t, days) Then
            nxt = False
        ElseIf t = "today" Then
            days.Add Date
        ElseIf t = "tomorrow" Or t = "tmrw" Or t = "tmw" Or t = "tmr" Then
            days.Add Date + 1
        ElseIf DateWord(t, dt) Then
            days.Add dt
        ElseIf NumberThenMonth(toks, i) Then
            days.Add MonthDay(MonthNum(toks(i + 1)), DayNum(t))      ' "26 nov"
            i = i + 1
        ElseIf MonthNum(t) > 0 And i < UBound(toks) Then
            dy = DayNum(toks(i + 1))                             ' "nov 26"
            If dy > 0 Then
                days.Add MonthDay(MonthNum(t), dy)
                i = i + 1
            End If
        ElseIf z <> "" Then
            zone = z
        ElseIf t = "after" Or t = "from" Then
            If t = "after" Then aft = True
        ElseIf t = "before" Or t = "by" Then
            bef = True
        ElseIf t = "am" Or t = "pm" Then
            If nT > 0 Then mer(nT) = IIf(t = "am", 1, 2)
        ElseIf Clock(t, m, mr) Then
            If nT < 4 Then
                nT = nT + 1
                tm(nT) = m
                mer(nT) = mr
            End If
        ElseIf t = "morning" Or t = "mornings" Then
            part = "morning"
        ElseIf t = "afternoon" Or t = "afternoons" Then
            part = "afternoon"
        ElseIf t = "evening" Or t = "evenings" Or t = "eod" Then
            part = "evening"
        ElseIf t = "anytime" Or t = "all" Or t = "any" Or t = "whenever" Then
            part = "all"
        ElseIf DayNum(t) > 0 And Right$(t, 2) Like "[a-z][a-z]" Then
            mo = 0                                               ' "the 26th", "26th nov"
            If i < UBound(toks) Then mo = MonthNum(toks(i + 1))
            days.Add MonthDay(mo, DayNum(t))
            If mo > 0 Then i = i + 1
        End If
        i = i + 1
    Loop

    If nT >= 2 Then
        FixRange tm(1), mer(1), tm(2), mer(2), sMin, eMin
        hasTime = True
    ElseIf nT = 1 And aft Then
        sMin = Guess(tm(1), mer(1))
        eMin = mHourTo
        If eMin <= sMin Then eMin = LATE_MIN
        hasTime = True
    ElseIf nT = 1 And bef Then
        sMin = mHourFrom
        eMin = Guess(tm(1), mer(1))
        If eMin <= sMin Then sMin = EARLY_MIN
        hasTime = True
    ElseIf nT = 1 Then
        sMin = Guess(tm(1), mer(1))                         ' a meeting starting then
        eMin = sMin + mLen
        hasTime = True
    ElseIf part <> "" Then
        Select Case part
            Case "morning": sMin = mHourFrom: eMin = 12 * 60
            Case "afternoon": sMin = 12 * 60: eMin = mHourTo
            Case "evening": sMin = 17 * 60: eMin = LATE_MIN
            Case Else: sMin = mHourFrom: eMin = mHourTo
        End Select
        If eMin <= sMin Then eMin = sMin + 60
        hasTime = True
    End If
    ParsePiece = hasTime Or days.Count > 0
End Function

' Minutes from midnight for a time as typed ("2", "2pm", "2:30", "14:00",
' "1430", "9a"); mer = 0 if no am/pm was given, 1 for am, 2 for pm.
Private Function Clock(ByVal t As String, ByRef mins As Long, ByRef mer As Long) As Boolean
    Dim h As Long, m As Long, p As Long, hh As String, mm As String
    mer = 0
    t = Replace(t, ".", ":")
    If Right$(t, 2) = "am" Then
        mer = 1
        t = Left$(t, Len(t) - 2)
    ElseIf Right$(t, 2) = "pm" Then
        mer = 2
        t = Left$(t, Len(t) - 2)
    ElseIf Right$(t, 1) = "a" Then
        mer = 1
        t = Left$(t, Len(t) - 1)
    ElseIf Right$(t, 1) = "p" Then
        mer = 2
        t = Left$(t, Len(t) - 1)
    End If
    If t = "" Then Exit Function
    p = InStr(t, ":")
    If p > 0 Then
        hh = Left$(t, p - 1)
        mm = Mid$(t, p + 1)
        If Len(hh) < 1 Or Len(hh) > 2 Or Len(mm) <> 2 Then Exit Function
        If Not AllDigits(hh) Or Not AllDigits(mm) Then Exit Function
        h = CLng(hh)
        m = CLng(mm)
    ElseIf AllDigits(t) Then
        If Len(t) <= 2 Then
            h = CLng(t)
        ElseIf Len(t) <= 4 Then
            h = CLng(Left$(t, Len(t) - 2))
            m = CLng(Right$(t, 2))
        Else
            Exit Function
        End If
    Else
        Exit Function
    End If
    If h > 23 Or m > 59 Then Exit Function
    If mer <> 0 And (h > 12 Or h = 0) Then Exit Function
    mins = h * 60 + m
    Clock = True
End Function

Private Function AllDigits(ByVal s As String) As Boolean
    Dim i As Long
    If s = "" Then Exit Function
    For i = 1 To Len(s)
        If Not Mid$(s, i, 1) Like "#" Then Exit Function
    Next
    AllDigits = True
End Function

' A time with am/pm applied: 12am is midnight, 12pm is noon.
Private Function WithMer(ByVal mins As Long, ByVal mer As Long) As Long
    Dim h As Long
    h = mins \ 60
    If mer = 1 And h = 12 Then mins = mins - 720
    If mer = 2 And h < 12 Then mins = mins + 720
    WithMer = mins
End Function

' A time typed without am/pm, read the way work times usually are:
' 7 to 11 = morning, 12 = noon, 1 to 6 = afternoon. 13:00 and up as typed.
Private Function Guess(ByVal mins As Long, ByVal mer As Long) As Long
    Dim h As Long
    If mer <> 0 Then
        Guess = WithMer(mins, mer)
        Exit Function
    End If
    h = mins \ 60
    If h >= 1 And h <= 6 Then mins = mins + 720
    Guess = mins
End Function

' Start and end of a range like "2-4pm", "11-1pm", "10am-2", "9-5".
Private Sub FixRange(ByVal a As Long, ByVal merA As Long, ByVal b As Long, ByVal merB As Long, _
                     ByRef sMin As Long, ByRef eMin As Long)
    If merA = 0 And merB <> 0 Then
        sMin = WithMer(a, merB)
        eMin = WithMer(b, merB)
        If sMin >= eMin Then sMin = WithMer(a, 3 - merB)             ' "11-1pm": 11 is morning
    ElseIf merA <> 0 And merB = 0 Then
        sMin = WithMer(a, merA)
        eMin = WithMer(b, merA)
        If eMin <= sMin Then eMin = WithMer(b, 3 - merA)             ' "10am-2": 2 is afternoon
    ElseIf merA = 0 And merB = 0 Then
        sMin = Guess(a, 0)
        eMin = Guess(b, 0)
        If eMin <= sMin And eMin < 720 Then eMin = eMin + 720
    Else
        sMin = WithMer(a, merA)
        eMin = WithMer(b, merB)
    End If
    If eMin > 1440 Then eMin = 1440
End Sub

' 1 = Monday ... 7 = Sunday, for a day's name or short name.
Private Function DayWord(ByVal t As String, ByRef wd As Long) As Boolean
    Select Case t
        Case "mon", "monday", "mondays": wd = 1
        Case "tue", "tues", "tuesday", "tuesdays": wd = 2
        Case "wed", "weds", "wednesday", "wednesdays": wd = 3
        Case "thu", "thur", "thurs", "thursday", "thursdays": wd = 4
        Case "fri", "friday", "fridays": wd = 5
        Case "sat", "saturday": wd = 6
        Case "sun", "sunday": wd = 7
        Case Else: Exit Function
    End Select
    DayWord = True
End Function

' "tue/wed/thu": each of those days.
Private Function DayList(ByVal t As String, ByVal days As Collection) As Boolean
    Dim parts() As String, i As Long, wd As Long
    If InStr(t, "/") = 0 Then Exit Function
    parts = Split(t, "/")
    For i = 0 To UBound(parts)
        If Not DayWord(parts(i), wd) Then Exit Function
    Next
    For i = 0 To UBound(parts)
        If DayWord(parts(i), wd) Then days.Add NextDay(wd, False)
    Next
    DayList = True
End Function

' The next such day, today included; "next" skips this week.
Private Function NextDay(ByVal wd As Long, ByVal nextWeek As Boolean) As Date
    Dim d As Date
    d = Date + ((wd - Weekday(Date, vbMonday) + 7) Mod 7)
    If nextWeek Then
        If d <= Date + (7 - Weekday(Date, vbMonday)) Then d = d + 7
    End If
    NextDay = d
End Function

' "11/26", "11/26/26", "11/26/2026".
Private Function DateWord(ByVal t As String, ByRef d As Date) As Boolean
    Dim parts() As String, mo As Long, dy As Long, y As Long
    If InStr(t, "/") = 0 Then Exit Function
    parts = Split(t, "/")
    If UBound(parts) < 1 Or UBound(parts) > 2 Then Exit Function
    If Not AllDigits(parts(0)) Or Not AllDigits(parts(1)) Then Exit Function
    mo = CLng(parts(0))
    dy = CLng(parts(1))
    If mo < 1 Or mo > 12 Or dy < 1 Or dy > 31 Then Exit Function
    If UBound(parts) = 2 Then
        If Not AllDigits(parts(2)) Then Exit Function
        y = CLng(parts(2))
        If y < 100 Then y = 2000 + y
        d = DateSerial(y, mo, dy)
    Else
        d = MonthDay(mo, dy)
    End If
    DateWord = (Day(d) = dy)
End Function

' Day mo/dy this year - or next year if that's well past. mo = 0: this month
' (or next month if that day has gone).
Private Function MonthDay(ByVal mo As Long, ByVal dy As Long) As Date
    Dim d As Date
    If mo = 0 Then
        d = DateSerial(Year(Date), Month(Date), dy)
        If d < Date Then d = DateSerial(Year(Date), Month(Date) + 1, dy)
    Else
        d = DateSerial(Year(Date), mo, dy)
        If d < Date - 60 Then d = DateSerial(Year(Date) + 1, mo, dy)
    End If
    MonthDay = d
End Function

Private Function MonthNum(ByVal t As String) As Long
    Select Case t
        Case "jan", "january": MonthNum = 1
        Case "feb", "february": MonthNum = 2
        Case "mar", "march": MonthNum = 3
        Case "apr", "april": MonthNum = 4
        Case "may": MonthNum = 5
        Case "jun", "june": MonthNum = 6
        Case "jul", "july": MonthNum = 7
        Case "aug", "august": MonthNum = 8
        Case "sep", "sept", "september": MonthNum = 9
        Case "oct", "october": MonthNum = 10
        Case "nov", "november": MonthNum = 11
        Case "dec", "december": MonthNum = 12
    End Select
End Function

' True for a day number followed by a month ("26 nov").
Private Function NumberThenMonth(toks() As String, ByVal i As Long) As Boolean
    If i >= UBound(toks) Then Exit Function
    If Not AllDigits(toks(i)) Then Exit Function
    If DayNum(toks(i)) = 0 Then Exit Function
    NumberThenMonth = (MonthNum(toks(i + 1)) > 0)
End Function

' 26 for "26", "26th", "1st", "2nd", "3rd"; 0 otherwise.
Private Function DayNum(ByVal t As String) As Long
    Dim suf As String
    suf = Right$(t, 2)
    If suf = "st" Or suf = "nd" Or suf = "rd" Or suf = "th" Then t = Left$(t, Len(t) - 2)
    If Not AllDigits(t) Or Len(t) > 2 Then Exit Function
    If CLng(t) >= 1 And CLng(t) <= 31 Then DayNum = CLng(t)
End Function

' Windows' name for a time zone written in an email ("ET", "PT", "London"...).
Private Function ZoneID(ByVal t As String) As String
    Select Case t
        Case "et", "est", "edt", "eastern", "ny", "nyc": ZoneID = "Eastern Standard Time"
        Case "ct", "cst", "cdt", "central", "chicago": ZoneID = "Central Standard Time"
        Case "mt", "mst", "mdt", "mountain", "denver": ZoneID = "Mountain Standard Time"
        Case "pt", "pst", "pdt", "pacific": ZoneID = "Pacific Standard Time"
        Case "gmt", "bst", "uk", "london": ZoneID = "GMT Standard Time"
        Case "cet", "cest", "paris", "frankfurt", "berlin": ZoneID = "W. Europe Standard Time"
        Case "hkt", "hk": ZoneID = "China Standard Time"
        Case "sgt", "singapore": ZoneID = "Singapore Standard Time"
        Case "jst", "tokyo": ZoneID = "Tokyo Standard Time"
    End Select
End Function

' A time in another time zone, as your local time (Outlook does the sums,
' summer time included).
Private Function ToLocal(ByVal t As Date, ByVal zone As String, ByRef converted As Boolean) As Date
    Dim tzs As Object, src As Object, dst As Object
    ToLocal = t
    On Error GoTo Done
    Set tzs = Application.TimeZones
    Set src = tzs.Item(zone)
    Set dst = tzs.CurrentTimeZone
    If src Is Nothing Or dst Is Nothing Then Exit Function
    If src.ID = dst.ID Then Exit Function
    ToLocal = tzs.ConvertTime(t, src, dst)
    converted = True
Done:
End Function

' The Hours box: "8am-6pm", "8-6", "7:30-19". False if it isn't a time range.
Private Function ParseHours(ByVal typed As String, ByRef fromMin As Long, ByRef toMin As Long) As Boolean
    Dim toks() As String, i As Long, tm(1 To 2) As Long, mer(1 To 2) As Long, n As Long
    Dim m As Long, mr As Long, s As String
    s = Replace(NormalizeTimes(typed), ",", " ")
    s = Replace(s, "-", " - ")
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    toks = Split(Trim$(s), " ")
    For i = 0 To UBound(toks)
        If toks(i) = "am" Or toks(i) = "pm" Then
            If n > 0 Then mer(n) = IIf(toks(i) = "am", 1, 2)
        ElseIf Clock(toks(i), m, mr) Then
            If n < 2 Then
                n = n + 1
                tm(n) = m
                mer(n) = mr
            End If
        End If
    Next
    If n < 2 Then Exit Function
    FixRange tm(1), mer(1), tm(2), mer(2), fromMin, toMin
    ParseHours = (toMin > fromMin)
End Function
