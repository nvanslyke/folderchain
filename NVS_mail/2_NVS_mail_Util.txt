' Folder list used by NVS_mail_FolderPicker (cached between uses).
Public gNVS_FolderCount As Long
Public gNVS_FolderID() As String
Public gNVS_FolderStoreID() As String
Public gNVS_FolderLabel() As String     ' indented name for the tree view
Public gNVS_FolderPath() As String      ' "Inbox\Projects\Acme" for filtering
Private mCacheKey As String

'==============================================================================
' Selection
'==============================================================================

Public Function NVS_IsMailLike(ByVal obj As Object) As Boolean
    If obj Is Nothing Then Exit Function
    If TypeOf obj Is Outlook.MailItem Then
        NVS_IsMailLike = True
    ElseIf TypeOf obj Is Outlook.MeetingItem Then
        NVS_IsMailLike = True
    End If
End Function

' The open email (if an email window is active) or the emails selected in
' the main window. If whole conversations are selected (collapsed headers in
' conversation view), their emails in the current folder are included.
Public Function NVS_GetSelectedItems() As Collection
    Dim col As New Collection, seen As Object, obj As Object, i As Long, j As Long
    Dim sel As Outlook.Selection, hdrs As Outlook.Selection
    Dim conv As Outlook.Conversation, cur As Outlook.Folder, it As Object
    Dim convItems As Collection

    Set seen = CreateObject("Scripting.Dictionary")
    On Error Resume Next

    If TypeName(Application.ActiveWindow) = "Inspector" Then
        Set obj = Application.ActiveInspector.CurrentItem
        If NVS_IsMailLike(obj) Then col.Add obj
        Set NVS_GetSelectedItems = col
        Exit Function
    End If

    Set sel = Application.ActiveExplorer.Selection
    If sel Is Nothing Then
        Set NVS_GetSelectedItems = col
        Exit Function
    End If

    For i = 1 To sel.Count
        Set obj = Nothing
        Set obj = sel.Item(i)
        If NVS_IsMailLike(obj) Then AddUnique col, seen, obj
    Next

    ' Collapsed conversation headers -> their emails in this folder.
    Set hdrs = Nothing
    Set hdrs = sel.GetSelection(olConversationHeaders)
    If Not hdrs Is Nothing Then
        Set cur = Application.ActiveExplorer.CurrentFolder
        For i = 1 To hdrs.Count
            Set conv = Nothing
            Set conv = hdrs.Item(i).GetConversation
            If Not conv Is Nothing And Not cur Is Nothing Then
                Set convItems = New Collection
                For j = 1 To conv.GetRootItems.Count
                    convItems.Add conv.GetRootItems.Item(j)
                Next
                If convItems.Count > 0 Then
                    For Each it In NVS_ConversationItems(convItems(1))
                        If NVS_IsMailLike(it) Then
                            If NVS_SameFolder(it.Parent, cur) Then AddUnique col, seen, it
                        End If
                    Next
                End If
            End If
        Next
    End If

    Set NVS_GetSelectedItems = col
End Function

Private Sub AddUnique(ByVal col As Collection, ByVal seen As Object, ByVal it As Object)
    Dim eid As String
    On Error Resume Next
    eid = it.EntryID
    On Error GoTo 0
    If eid = "" Then
        col.Add it
    ElseIf Not seen.Exists(eid) Then
        seen.Add eid, True
        col.Add it
    End If
End Sub

Public Function NVS_SameFolder(ByVal a As Object, ByVal b As Object) As Boolean
    On Error Resume Next
    If a Is Nothing Or b Is Nothing Then Exit Function
    NVS_SameFolder = Application.Session.CompareEntryIDs(a.EntryID, b.EntryID)
End Function

' Folder picker

' Shows NVS_mail_FolderPicker with defaultFolder pre-selected. Nothing = cancelled.
Public Function NVS_PickFolder(ByVal title As String, ByVal info As String, _
                              ByVal defaultFolder As Outlook.Folder, _
                              Optional ByVal okCaption As String = "OK") As Outlook.Folder
    Dim frm As NVS_mail_FolderPicker
    Set frm = New NVS_mail_FolderPicker
    frm.Prepare title, info, defaultFolder, okCaption
    frm.Show
    If Not frm.Cancelled Then Set NVS_PickFolder = frm.SelectedFolder
    Unload frm
End Function

' Builds the folder list for the default mailbox plus any extra stores
' (e.g. the shared mailbox the current email lives in).
Public Sub NVS_BuildFolderCache(ByVal extraStores As Collection, ByVal force As Boolean)
    Dim stores As New Collection, st As Object, key As String, v As Variant

    AddStore stores, Application.Session.DefaultStore
    If Not extraStores Is Nothing Then
        For Each v In extraStores
            AddStore stores, v
        Next
    End If

    For Each st In stores
        key = key & "|" & st.StoreID
    Next
    If Not force And key = mCacheKey And gNVS_FolderCount > 0 Then Exit Sub

    gNVS_FolderCount = 0
    ReDim gNVS_FolderID(0 To 255)
    ReDim gNVS_FolderStoreID(0 To 255)
    ReDim gNVS_FolderLabel(0 To 255)
    ReDim gNVS_FolderPath(0 To 255)

    For Each st In stores
        On Error Resume Next
        AddFolderTree st.GetRootFolder, 0, st.StoreID, st.DisplayName, (st.StoreID = Application.Session.DefaultStore.StoreID)
        On Error GoTo 0
    Next
    mCacheKey = key
End Sub

Private Sub AddStore(ByVal stores As Collection, ByVal st As Object)
    Dim s As Object
    If st Is Nothing Then Exit Sub
    For Each s In stores
        If s.StoreID = st.StoreID Then Exit Sub
    Next
    stores.Add st
End Sub

Private Sub AddFolderTree(ByVal f As Outlook.Folder, ByVal depth As Long, ByVal storeID As String, _
                          ByVal storeName As String, ByVal isDefaultStore As Boolean)
    Dim subs As Outlook.Folders, i As Long, rel As String, p As Long
    On Error GoTo Done

    If depth > 0 Then
        If f.DefaultItemType <> olMailItem Then Exit Sub
        If IsHiddenFolder(f) Then Exit Sub
    End If

    ' "\\mailbox\Inbox\Projects" -> "Inbox\Projects"
    rel = Mid$(f.FolderPath, 3)
    p = InStr(rel, "\")
    If p > 0 Then rel = Mid$(rel, p + 1) Else rel = storeName
    If Not isDefaultStore And depth > 0 Then rel = rel & "   [" & storeName & "]"

    If gNVS_FolderCount > UBound(gNVS_FolderID) Then
        ReDim Preserve gNVS_FolderID(0 To gNVS_FolderCount * 2)
        ReDim Preserve gNVS_FolderStoreID(0 To gNVS_FolderCount * 2)
        ReDim Preserve gNVS_FolderLabel(0 To gNVS_FolderCount * 2)
        ReDim Preserve gNVS_FolderPath(0 To gNVS_FolderCount * 2)
    End If
    gNVS_FolderID(gNVS_FolderCount) = f.EntryID
    gNVS_FolderStoreID(gNVS_FolderCount) = storeID
    If depth = 0 Then
        gNVS_FolderLabel(gNVS_FolderCount) = storeName
    Else
        gNVS_FolderLabel(gNVS_FolderCount) = String$(depth * 4, " ") & f.Name
    End If
    gNVS_FolderPath(gNVS_FolderCount) = rel
    gNVS_FolderCount = gNVS_FolderCount + 1

    Set subs = f.Folders
    For i = 1 To subs.Count
        AddFolderTree subs.Item(i), depth + 1, storeID, storeName, isDefaultStore
    Next
Done:
End Sub

Private Function IsHiddenFolder(ByVal f As Outlook.Folder) As Boolean
    On Error Resume Next
    IsHiddenFolder = CBool(f.PropertyAccessor.GetProperty(PR_ATTR_HIDDEN))
End Function

Public Function NVS_FolderCacheIndex(ByVal entryID As String) As Long
    Dim i As Long
    NVS_FolderCacheIndex = -1
    If entryID = "" Then Exit Function
    For i = 0 To gNVS_FolderCount - 1
        If gNVS_FolderID(i) = entryID Then
            NVS_FolderCacheIndex = i
            Exit Function
        End If
    Next
End Function

Public Function NVS_FolderFromCache(ByVal index As Long) As Outlook.Folder
    On Error Resume Next
    If index < 0 Or index >= gNVS_FolderCount Then Exit Function
    Set NVS_FolderFromCache = Application.Session.GetFolderFromID(gNVS_FolderID(index), gNVS_FolderStoreID(index))
End Function

' Remembers / recalls the last folder mail was moved to with NVS_mail.
Public Sub NVS_SetLastMoveFolder(ByVal f As Outlook.Folder)
    On Error Resume Next
    SaveSetting NVS_APP, "Move", "LastFolderID", f.EntryID
    SaveSetting NVS_APP, "Move", "LastStoreID", f.StoreID
End Sub

Public Function NVS_GetLastMoveFolder() As Outlook.Folder
    Dim eid As String, sid As String
    eid = GetSetting(NVS_APP, "Move", "LastFolderID", "")
    sid = GetSetting(NVS_APP, "Move", "LastStoreID", "")
    If eid = "" Then Exit Function
    On Error Resume Next
    Set NVS_GetLastMoveFolder = Application.Session.GetFolderFromID(eid, sid)
End Function

Public Function NVS_AddressEntrySmtp(ByVal ae As Outlook.AddressEntry) As String
    Dim s As String
    On Error Resume Next
    If ae Is Nothing Then Exit Function
    Select Case ae.AddressEntryUserType
        Case olExchangeUserAddressEntry, olExchangeRemoteUserAddressEntry
            s = ae.GetExchangeUser.PrimarySmtpAddress
        Case olExchangeDistributionListAddressEntry
            s = ae.GetExchangeDistributionList.PrimarySmtpAddress
    End Select
    If s = "" Then s = ae.PropertyAccessor.GetProperty(PR_SMTP_ADDRESS)
    If s = "" And InStr(ae.Address, "@") > 0 Then s = ae.Address
    NVS_AddressEntrySmtp = s
End Function

Public Function NVS_RecipientSmtp(ByVal r As Outlook.Recipient) As String
    Dim s As String
    On Error Resume Next
    s = r.PropertyAccessor.GetProperty(PR_SMTP_ADDRESS)
    If s = "" Then s = NVS_AddressEntrySmtp(r.AddressEntry)
    If s = "" And InStr(r.Address, "@") > 0 Then s = r.Address
    If s = "" Then s = r.Name
    NVS_RecipientSmtp = s
End Function

Public Function NVS_SenderSmtp(ByVal it As Object) As String
    Dim s As String, ae As Outlook.AddressEntry
    On Error Resume Next
    If it.SenderEmailType = "SMTP" Then s = it.SenderEmailAddress
    If s = "" Then
        Set ae = it.Sender
        If Not ae Is Nothing Then s = NVS_AddressEntrySmtp(ae)
    End If
    If s = "" And InStr(it.SenderEmailAddress, "@") > 0 Then s = it.SenderEmailAddress
    NVS_SenderSmtp = s
End Function

' Your own addresses (all accounts), so they can be left off invites.
Public Function NVS_MyAddresses() As Object
    Dim d As Object, i As Long, s As String
    Set d = CreateObject("Scripting.Dictionary")
    d.CompareMode = vbTextCompare
    On Error Resume Next
    For i = 1 To Application.Session.Accounts.Count
        s = ""
        s = Application.Session.Accounts.Item(i).SmtpAddress
        If s <> "" And Not d.Exists(s) Then d.Add s, True
    Next
    s = ""
    s = NVS_AddressEntrySmtp(Application.Session.CurrentUser.AddressEntry)
    If s <> "" And Not d.Exists(s) Then d.Add s, True
    s = ""
    s = Application.Session.CurrentUser.Name
    If s <> "" And Not d.Exists(s) Then d.Add s, True
    Set NVS_MyAddresses = d
End Function

' Replaces characters Windows doesn't allow in file names.
Public Function NVS_SafeFileName(ByVal s As String, Optional ByVal maxLen As Long = 120) As String
    Dim bad As Variant, i As Long
    bad = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For i = LBound(bad) To UBound(bad)
        s = Replace(s, bad(i), "_")
    Next
    For i = 0 To 31
        s = Replace(s, Chr$(i), " ")
    Next
    s = Trim$(s)
    If Len(s) > maxLen Then s = Trim$(Left$(s, maxLen))
    Do While Len(s) > 0 And (Right$(s, 1) = "." Or Right$(s, 1) = " ")
        s = Left$(s, Len(s) - 1)
    Loop
    NVS_SafeFileName = s
End Function

Public Sub NVS_SplitExt(ByVal fileName As String, ByRef baseName As String, ByRef ext As String)
    Dim p As Long
    p = InStrRev(fileName, ".")
    If p > 1 Then
        baseName = Left$(fileName, p - 1)
        ext = Mid$(fileName, p)
    Else
        baseName = fileName
        ext = ""
    End If
End Sub

' folder\base.ext, or folder\base (2).ext etc. if that already exists.
' Keeps the full path under Windows' 260 character limit.
Public Function NVS_UniquePath(ByVal folder As String, ByVal baseName As String, ByVal ext As String) As String
    Dim fso As Object, candidate As String, n As Long, room As Long
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Right$(folder, 1) = "\" Then folder = Left$(folder, Len(folder) - 1)

    room = 250 - Len(folder) - 1 - Len(ext) - 5
    If room < 10 Then room = 10
    If Len(baseName) > room Then baseName = Trim$(Left$(baseName, room))
    If baseName = "" Then baseName = "file"

    candidate = folder & "\" & baseName & ext
    n = 1
    Do While fso.FileExists(candidate) Or fso.FolderExists(candidate)
        n = n + 1
        candidate = folder & "\" & baseName & " (" & n & ")" & ext
    Loop
    NVS_UniquePath = candidate
End Function

' True for pictures embedded in the body (signature logos etc.).
Public Function NVS_IsInlineAttachment(ByVal att As Outlook.Attachment, ByVal htmlBody As String) As Boolean
    Dim cid As String, hidden As Boolean
    On Error Resume Next
    hidden = att.PropertyAccessor.GetProperty(PR_ATTACHMENT_HIDDEN)
    If hidden Then NVS_IsInlineAttachment = True: Exit Function
    cid = att.PropertyAccessor.GetProperty(PR_ATTACH_CONTENT_ID)
    If cid <> "" And htmlBody <> "" Then
        If InStr(1, htmlBody, "cid:" & cid, vbTextCompare) > 0 Then NVS_IsInlineAttachment = True
    End If
End Function
